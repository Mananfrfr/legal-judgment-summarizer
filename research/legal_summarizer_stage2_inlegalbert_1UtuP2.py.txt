    LABEL2ID = {label: i for i, label in enumerate(LABELS)}

    def json_to_df(documents, split):
        rows, skipped, multiple_sets = [], [], []
        for doc_index, doc in enumerate(documents):
            annotations = doc.get('annotations') or []
            # A document may contain labels from several annotators. Never merge their
            # results: this would duplicate offsets and may invent contradictory labels.
            candidates = []
            for annotation_index, annotation in enumerate(annotations):
                if not isinstance(annotation, dict):
                    continue
                results = annotation.get('result') or []
                spans_here = [r['value'] for r in results if isinstance(r, dict)
                              and isinstance(r.get('value'), dict)
                              and 'text' in r['value']]
                if spans_here:
                    candidates.append((annotation_index, spans_here, annotation))

            if not candidates:
                skipped.append(doc_index)
                continue

            # Prefer an explicitly adjudicated/ground-truth annotation, if supplied.
            # The Hugging Face Parquet schema often contains only 'result' and lacks
            # adjudication metadata; in that case, use a documented deterministic policy.
            adjudicated = [item for item in candidates if
                           item[2].get('ground_truth') is True or
                           item[2].get('is_adjudicated') is True]
            if len(adjudicated) > 1:
                raise ValueError(f'{split} document {doc_index}: more than one annotation '
                                 'is marked adjudicated; review the source data.')
            selected_idx, spans, _ = (adjudicated[0] if adjudicated else candidates[0])
            selection_reason = 'adjudicated' if adjudicated else 'first_nonempty'

            if len(annotations) > 1:
                # Compare the entire labeled sentence sequence across nonempty sets.
                # Keep only counts/indices in audit, never duplicate the document text.
                def signature(annotator_spans):
                    records = [(v.get('start'), v.get('end'), v.get('text'),
                                tuple(v.get('labels') or ())) for v in annotator_spans]
                    return tuple(sorted(records, key=lambda x: (str(x[0]), str(x[1]), str(x[2]))))
                disagree = len({signature(item[1]) for item in candidates}) > 1
                multiple_sets.append(dict(doc_index=doc_index,
                                          total_annotation_sets=len(annotations),
                                          nonempty_annotation_sets=len(candidates),
                                          selected_annotation_index=selected_idx,
                                          selection_reason=selection_reason,
                                          annotation_sets_disagree=disagree))
            for value in spans:
                if not isinstance(value.get('start'), int) or not isinstance(value.get('end'), int):
                    raise ValueError(f'{split} document {doc_index}: missing sentence offsets')
                if value['end'] <= value['start'] or not value['text'].strip():
                    raise ValueError(f'{split} document {doc_index}: invalid or empty span')
                labels = value.get('labels') or []
                if len(labels) > 1:
                    raise ValueError('Expected one rhetorical role per sentence')
                label = labels[0] if labels else None
                if label is not None and label not in LABEL2ID:
                    raise ValueError(f'Unknown label: {label}')
                rows.append(dict(split=split, doc_index=doc_index, annotation_index=selected_idx, meta=doc.get('meta'),
                                 start=value['start'], end=value['end'], text=value['text'], label=label))
        df = pd.DataFrame(rows, columns=['split','doc_index','annotation_index','meta','start','end','text','label'])
        df = df.sort_values(['doc_index', 'start', 'end']).reset_index(drop=True)
        if df.duplicated(['doc_index', 'start', 'end']).any():
            raise ValueError(f'{split}: duplicate sentence spans')
        df['sentence_index'] = df.groupby('doc_index').cumcount()
        report = dict(raw_documents=len(documents), represented_documents=int(df.doc_index.nunique()),
                      sentences=len(df), skipped_document_indices=skipped,
                      unlabeled=int(df.label.isna().sum()),
                      multi_annotation_documents=len(multiple_sets),
                      multi_annotation_disagreements=sum(x['annotation_sets_disagree'] for x in multiple_sets),
                      annotation_selection_policy='adjudicated_if_marked_else_first_nonempty',
                      multi_annotation_details=multiple_sets)
        return df, report

    frames, audit = {}, {}

        return {hashlib.sha256(' '.join(' '.join(g.text).split()).encode()).hexdigest()
                for _, g in df.groupby('doc_index')}

        if overlap:
            raise ValueError(f'{a}/{b}: {len(overlap)} identical documents; resolve split leakage first')

              f'{info["unlabeled"]} unlabeled sentences')

@torch.no_grad()
def predict_labels(model, loader):
    model.eval()
    predicted_ids, true_ids = [], []
    for batch in tqdm(loader, desc='Evaluating', leave=False):
        gold = batch.pop('labels')
        inputs = {k:v.to(DEVICE) for k,v in batch.items()}
        logits = model(**inputs).logits
        predicted_ids.extend(logits.argmax(dim=-1).cpu().tolist())
        true_ids.extend(gold.tolist())
    return np.array(true_ids), np.array(predicted_ids)

def score_ids(y_true, y_pred):
    all_ids = list(range(len(LABELS)))
    return {'accuracy':accuracy_score(y_true,y_pred),
            'macro_f1':f1_score(y_true,y_pred,labels=all_ids,average='macro',zero_division=0),
            'weighted_f1':f1_score(y_true,y_pred,labels=all_ids,average='weighted',zero_division=0)}

    total_steps = steps_per_epoch * CONFIG['epochs']
    scheduler = get_linear_schedule_with_warmup(

        num_training_steps=total_steps
    )

    for epoch in range(1,CONFIG['epochs']+1):
        model.train()
        batch_losses = []
        bar = tqdm(train_loader,desc=f'Train epoch {epoch}/{CONFIG["epochs"]}')
        n_batches = len(train_loader)
        for step,batch in enumerate(bar):
            inputs = {k:v.to(DEVICE) for k,v in batch.items()}
            # Last accumulation group may contain fewer batches.
            group_start = (step // CONFIG['grad_accum_steps'])*CONFIG['grad_accum_steps']
            group_size = min(CONFIG['grad_accum_steps'],n_batches-group_start)
            with torch.amp.autocast('cuda', enabled=use_amp):
                loss = model(**inputs).loss
            if not torch.isfinite(loss).item():
                raise RuntimeError('Loss is NaN or infinite; check data and learning rate.')
            scaler.scale(loss/group_size).backward()
            batch_losses.append(float(loss.item()))
            if ((step+1)%CONFIG['grad_accum_steps']==0) or (step+1==n_batches):
                scaler.unscale_(opt)
                torch.nn.utils.clip_grad_norm_(model.parameters(),CONFIG['max_grad_norm'])
                scaler.step(opt)
                scaler.update()
                scheduler.step()
                opt.zero_grad(set_to_none=True)
            bar.set_postfix(loss=f'{np.mean(batch_losses[-20:]):.3f}')
        true_ids, pred_ids = predict_labels(model,dev_loader)

def restore_stage2_zip(zip_path):
    """Safely extract our *flat-root* artifacts ZIP into ART (not /content)."""
    zip_path = Path(zip_path)
    if not zip_path.exists():
        print('ZIP not found:', zip_path)
        return False
    with zipfile.ZipFile(zip_path) as zipped:
        for member in zipped.infolist():
            if member.is_dir():
                continue
            path = PurePosixPath(member.filename.replace('\\', '/'))
            if path.is_absolute() or '..' in path.parts:
                raise ValueError('Unsafe archive member: ' + member.filename)
            # Accept either a flat archive or one rooted at stage2_artifacts/.
            parts = list(path.parts)
            if parts and parts[0] == 'stage2_artifacts':
                parts = parts[1:]
            if not parts:
                continue
            target = ART.joinpath(*parts)
            if not target.resolve().is_relative_to(ART.resolve()):
                raise ValueError('Archive member escapes artifact directory')
            target.parent.mkdir(parents=True, exist_ok=True)
            with zipped.open(member) as source, target.open('wb') as dest:
                import shutil
                shutil.copyfileobj(source, dest)
    print('Restored ZIP into:', ART)
    return True

from pathlib import Path
from IPython.display import display
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

ART = Path('/content/stage2_artifacts' if Path('/content').exists() else 'stage2_artifacts')
PRED_FILE = ART / 'inlegalbert_dev_predictions.csv'
REPORT_FILE = ART / 'inlegalbert_dev_class_report.csv'
CONFUSION_FILE = ART / 'inlegalbert_dev_confusion.csv'

required_files = [PRED_FILE, REPORT_FILE, CONFUSION_FILE]
missing = [f.name for f in required_files if not f.is_file()]

if missing:
    print('ERROR ANALYSIS NOT RUN: these saved validation files are missing:', missing)
    print('Run the Stage 2.1A RECOVERY cell above. If you have the saved ZIP,')
    print('run the optional upload cell, rerun recovery, and then rerun error analysis.')
    print('Without a ZIP or checkpoint, you must recreate dev predictions by training + evaluation.')
    print('Do not enable RUN_FINAL_TEST for this analysis.')
else:
    dev_predictions = pd.read_csv(PRED_FILE)
    class_report = pd.read_csv(REPORT_FILE, index_col=0)
    confusion = pd.read_csv(CONFUSION_FILE, index_col=0)
    roles = list(confusion.index)
    assert confusion.columns.tolist() == roles, 'Confusion matrix row and column labels differ.'
    assert {'text', 'label', 'prediction'}.issubset(dev_predictions.columns), 'Missing prediction CSV columns.'
    assert set(roles).issubset(class_report.index), 'Class report missing one or more labels.'

    # A. Worst-performing classes (F1 = balance of precision and recall).
    per_class = class_report.loc[roles, ['precision','recall','f1-score','support']].copy()
    per_class[['precision','recall','f1-score','support']] = per_class[
        ['precision','recall','f1-score','support']].apply(pd.to_numeric, errors='coerce')
    per_class = per_class.sort_values(['f1-score','support'], ascending=[True,True])
    per_class.to_csv(ART / 'error_analysis_class_ranking.csv')
    print('\n1) CLASS-WISE PERFORMANCE — LOWEST F1 FIRST')
    display(per_class.round(3))

    # B. Most common off-diagonal (wrong) predictions: actual -> predicted.
    wrong_pairs = []
    for actual in roles:
        for predicted in roles:
            if actual != predicted:
                n = int(confusion.loc[actual, predicted])
                if n > 0:
                    wrong_pairs.append({'actual_role':actual, 'predicted_role':predicted,
                                        'mistakes':n, 'actual_role_total':int(confusion.loc[actual].sum())})
    mistakes = pd.DataFrame(wrong_pairs).sort_values('mistakes', ascending=False)
    mistakes['percent_of_actual_role'] = (100 * mistakes['mistakes'] /
                                          mistakes['actual_role_total']).round(1)
    mistakes.to_csv(ART / 'error_analysis_confused_pairs.csv', index=False)
    print('\n2) TOP 12 CONFUSED LABEL PAIRS (TRUE → PREDICTED)')
    display(mistakes.head(12))

    # C. Normalized confusion matrix — each row sums to 100% for a given true role.
    cm = confusion.to_numpy(dtype=float)
    totals = cm.sum(axis=1, keepdims=True)
    percentages = np.divide(cm, totals, out=np.zeros_like(cm), where=totals != 0) * 100
    fig, ax = plt.subplots(figsize=(13, 10))
    im = ax.imshow(percentages, cmap='Blues', vmin=0, vmax=100)
    ax.set_xticks(np.arange(len(roles)), labels=roles, rotation=55, ha='right')
    ax.set_yticks(np.arange(len(roles)), labels=roles)
    ax.set_xlabel('Predicted rhetorical role')
    ax.set_ylabel('Actual (gold) rhetorical role')
    ax.set_title('InLegalBERT: dev confusion matrix (% within each actual class)')
    for i in range(len(roles)):
        for j in range(len(roles)):
            if cm[i,j] > 0:
                ax.text(j, i, f'{percentages[i,j]:.0f}%', ha='center', va='center',
                        fontsize=8, color='white' if percentages[i,j] >= 55 else 'black')
    fig.colorbar(im, ax=ax, label='Percent of actual-role examples')
    fig.tight_layout()
    fig.savefig(ART / 'error_analysis_normalized_confusion.png', dpi=170, bbox_inches='tight')
    plt.show()

    # D. Real dev errors for the three largest confusion pairs; keep document context IDs.
    bad = dev_predictions.loc[dev_predictions['label'] != dev_predictions['prediction']].copy()
    bad['short_text'] = (bad['text'].fillna('').astype(str)
                         .str.replace(r'\s+', ' ', regex=True).str.slice(0, 240))
    example_frames = []
    for _, pair in mistakes.head(3).iterrows():
        group = bad.loc[(bad.label == pair.actual_role) &
                        (bad.prediction == pair.predicted_role)].head(4)
        example_frames.append(group)
    selected_examples = pd.concat(example_frames, ignore_index=True) if example_frames else bad.head(0)
    example_cols = [c for c in ['doc_index','sentence_index','label','prediction','short_text']
                    if c in selected_examples.columns]
    selected_examples[example_cols].to_csv(ART / 'error_analysis_examples.csv', index=False)
    print('\n3) REAL MISCLASSIFIED VALIDATION SENTENCES (up to four per top three pairs)')
    display(selected_examples[example_cols])

    # E. Simple overview and interpretation notes.
    correct_count = int((dev_predictions['label'] == dev_predictions['prediction']).sum())
    print(f'\n4) OVERVIEW: {correct_count}/{len(dev_predictions)} dev sentences correctly classified')
    print(f'   Misclassified dev sentences: {len(bad)}')
    print('   Lowest-F1 labels:', ', '.join(per_class.head(3).index.tolist()))
    print('   Biggest confusion:', f"{mistakes.iloc[0].actual_role} → {mistakes.iloc[0].predicted_role}")
    print('Interpret errors with care: one development judgment has conflicting human annotations.')
    print('Saved three CSV reports + one chart to:', ART)
