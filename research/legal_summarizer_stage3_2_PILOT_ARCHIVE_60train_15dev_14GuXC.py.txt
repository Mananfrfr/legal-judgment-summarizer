def checkpoint_signature(path):
    h=hashlib.sha256()
    for name in ('config.json','tokenizer.json','model.safetensors'):
        p=Path(path)/name
        if not p.is_file(): raise FileNotFoundError(p)
        h.update(name.encode());h.update(str(p.stat().st_size).encode())
        with p.open('rb') as f:
            h.update(f.read(128*1024))
            if p.stat().st_size>128*1024:
                f.seek(-128*1024,2);h.update(f.read(128*1024))
    return h.hexdigest()

class SentenceBiLSTM(nn.Module):
    # Exactly the same architecture as saved Stage 2.3 Softmax head
    def __init__(self,in_dim=768,n_labels=13,hidden=128,dropout=0.3):
        super().__init__()
        self.norm=nn.LayerNorm(in_dim)
        self.drop=nn.Dropout(dropout)
        self.lstm=nn.LSTM(in_dim,hidden,num_layers=1,batch_first=True,bidirectional=True)
        self.classifier=nn.Linear(hidden*2,n_labels)
    def forward(self,x,lengths):
        x=self.drop(self.norm(x))
        packed=nn.utils.rnn.pack_padded_sequence(x,lengths.cpu(),batch_first=True,enforce_sorted=False)
        packed_out,_=self.lstm(packed)
        output,_=nn.utils.rnn.pad_packed_sequence(packed_out,batch_first=True,total_length=x.shape[1])
        return self.classifier(self.drop(output))

def pack_context_ids(target_ids,previous_ids,next_ids,max_length=256,target_cap=192):
    tgt=list(target_ids[:min(target_cap,max_length-3)])
    budget=max_length-len(tgt)-3
    pmark=MARK_PREV if previous_ids else []
    nmark=MARK_NEXT if next_ids else []
    free=budget-len(pmark)-len(nmark)
    if free<0: raise ValueError('No token budget for context markers')
    prev_cap=min(len(previous_ids),(free+1)//2) if previous_ids else 0
    next_cap=min(len(next_ids),free//2) if next_ids else 0
    left=free-prev_cap-next_cap
    prev_cap+=min(len(previous_ids)-prev_cap,left)
    left=free-prev_cap-next_cap
    next_cap+=min(len(next_ids)-next_cap,left)
    prev_part=list(previous_ids[-prev_cap:]) if prev_cap else []
    next_part=list(next_ids[:next_cap]) if next_cap else []
    context=pmark+prev_part+nmark+next_part
    ids=[TOKENIZER.cls_token_id]+tgt+[TOKENIZER.sep_token_id]+context+[TOKENIZER.sep_token_id]
    segment0=len(tgt)+2
    item={'input_ids':ids,'attention_mask':[1]*len(ids),
          'token_type_ids':[0]*segment0+[1]*(len(ids)-segment0)}
    assert len(ids)<=max_length
    return item

def dataset_tail(path, dataset_name):
    parts=path.replace('\\','/').split('/')
    aliases={dataset_name.lower(), dataset_name.lower().replace('-','')}
    hits=[i for i,p in enumerate(parts) if p.lower().replace('_','-') in aliases]
    return parts[hits[-1]+1:] if hits else None

def index_official_dataset(z):
    inabs={'train':{'judgement':{},'summary':{}},'test':{'judgement':{},'summary':{}}}
    inext={}
    for info in z.infolist():
        if info.is_dir() or not info.file_size or info.file_size>20_000_000: continue
        rel=dataset_tail(info.filename,'IN-Abs')
        if rel and len(rel)>=3:
            low=[x.lower() for x in rel]
            split='train' if low[0]=='train-data' else 'test' if low[0]=='test-data' else None
            typ=low[1]
            if split and typ in ('judgement','judgment','summary'):
                typ='judgement' if typ=='judgment' else typ
                key='/'.join(x.lower() for x in rel[2:])
                if key.endswith('.txt'):
                    inabs[split][typ][key]=info.filename
        rel=dataset_tail(info.filename,'IN-Ext')
        if rel and len(rel)>=2 and rel[0].lower() in ('judgement','judgment'):
            key='/'.join(x.lower() for x in rel[1:])
            if key.endswith('.txt'): inext[key]=info.filename
    pairs={}
    for split in ('train','test'):
        d=inabs[split]
        keys=sorted(set(d['judgement']) & set(d['summary']))
        pairs[split]=[{'id':key,'judgment_path':d['judgement'][key],
                       'summary_path':d['summary'][key], 'split':split} for key in keys]
    return pairs, inext, inabs

def clean_text(raw):
    return raw.decode('utf-8',errors='replace') if isinstance(raw,bytes) else str(raw)

def normalized_doc(s):
    return re.sub(r'[^a-z0-9]+',' ',s.lower()).strip()

def document_signatures(s):
    s=normalized_doc(s)
    h=lambda x: hashlib.sha256(x.encode('utf-8')).hexdigest()
    return (h(s),h(s[:4000]),len(s))

def normalize_for_segmentation(text):
    text=text.replace('\u00ad','')
    text=re.sub(r'(?<=\w)-\s*\n\s*(?=\w)','',text)
    return re.sub(r'\s+',' ',text).strip()

def split_sentences(text):
    clean=normalize_for_segmentation(text)
    NLP.max_length=max(NLP.max_length,len(clean)+100)
    return [s.text.strip() for s in NLP(clean).sents if re.search('[A-Za-z]',s.text)]

def select_window_indices(n,limit):
    if n<=limit:return list(range(n))
    first=round(limit*0.5);last=round(limit*0.25);middle=limit-first-last
    mid_start=max(first, (n-middle)//2)
    result=list(range(first))+list(range(mid_start,mid_start+middle))+list(range(n-last,n))
    assert len(result)==len(set(result))==limit
    return sorted(result)

def weak_sentence_targets(sentences,reference,positive_frac=POSITIVE_FRACTION):
    assert len(sentences)>0
    tfidf=TfidfVectorizer(stop_words='english',ngram_range=(1,2),max_features=12000,
                          token_pattern=r'(?u)\b\w\w+\b')
    try:
        vectors=tfidf.fit_transform(list(sentences)+[reference]);v=vectors[:-1]
        similarities=cosine_similarity(v,vectors[-1]).ravel()
    except ValueError:
        similarities=np.zeros(len(sentences),dtype=np.float32)
    eligible=[i for i,s in enumerate(sentences) if len(s.split())>=5]
    npos=min(len(eligible),max(1,round(len(eligible)*positive_frac)))
    rank=sorted(eligible,key=lambda i:(-similarities[i],i))
    result=np.zeros(len(sentences),dtype=np.float32)
    result[rank[:npos]]=1
    return result

            reference=clean_text(z.read(row['summary_path']))
            all_sents=split_sentences(judgment)
            if len(all_sents)<5:raise ValueError('Too few sentences in '+row['id'])
            kept=select_window_indices(len(all_sents),MAX_SENTENCES_PER_DOC)
            sents=[all_sents[i] for i in kept]
            sample={'id':row['id'],'sentences':sents,'original_indices':kept,
                    'reference':reference,'original_sentence_count':len(all_sents),
                    'retained_sentence_count':len(sents),'input_words':len(judgment.split())}
            if make_targets:sample['pseudo_y']=weak_sentence_targets(sents,reference)
            result.append(sample)
    return result

@torch.inference_mode()
def encode_case(case):
    sents=case['sentences']
    raw=TOKENIZER(sents,add_special_tokens=False,padding=False,return_attention_mask=False)['input_ids']
    original=case['original_indices']
    # Never pretend the first/last sentences of distant sampled windows are adjacent.
    inputs=[pack_context_ids(raw[i],
                 raw[i-1] if i and original[i-1]+1==original[i] else [],
                 raw[i+1] if i+1<len(sents) and original[i]+1==original[i+1] else [],
                 max_length=256,target_cap=192)
            for i in range(len(sents))]
    vec=[]
    for start in range(0,len(inputs),ENCODER_BATCH_SIZE):
        batch=COLLATOR(inputs[start:start+ENCODER_BATCH_SIZE])
        inputs_on_device={k:v.to(DEVICE) for k,v in batch.items()}
        with torch.amp.autocast(device_type='cuda',enabled=(DEVICE.type=='cuda')):
            x=ENCODER.base_model(**inputs_on_device,return_dict=True).last_hidden_state[:,0,:]
        vec.append(x.float().cpu().to(torch.float16))
    # Mimic original Stage 2.3 path: float16 vectors on disk -> float32 BiLSTM.
    encoded=torch.cat(vec,dim=0).contiguous()
    # The Stage 2.3 BiLSTM must reset at each contiguous original-document span.
    starts=[0]+[i for i in range(1,len(sents)) if original[i]!=original[i-1]+1]
    ends=starts[1:]+[len(sents)]
    pred=[]
    for lo,hi in zip(starts,ends):
        seq=encoded[lo:hi].float().unsqueeze(0).to(DEVICE)
        logits=HEAD(seq,torch.tensor([hi-lo],dtype=torch.long))[0]
        pred.append(logits.float().softmax(dim=-1).cpu().numpy().astype(np.float16))
    probs=np.concatenate(pred,axis=0)
    return encoded.numpy(),probs

def verified_feature_cache(case):
    digest=sentences_digest(case['sentences'])
    digest=hashlib.sha256((digest+json.dumps(case['original_indices'])).encode('utf-8')).hexdigest()
    key=hashlib.sha256((case['id']+SIGNATURE+digest).encode()).hexdigest()[:28]
    cache=FEATURE_CACHE/(key+'.npz')
    if cache.is_file() and not FORCE_REENCODE:
        with np.load(cache,allow_pickle=False) as d:
            if (str(d['sig'])==SIGNATURE and str(d['doc_digest'])==digest
                 and tuple(d['cls'].shape)==(len(case['sentences']),768)
                 and tuple(d['roles'].shape)==(len(case['sentences']),13)):
                return d['cls'].astype(np.float32), d['roles'].astype(np.float32)
        print('Cache validation failed; re-encoding:',case['id'])
    embeddings,roles=encode_case(case)
    assert embeddings.shape==(len(case['sentences']),768)
    assert roles.shape==(len(case['sentences']),13) and np.isfinite(embeddings).all() and np.isfinite(roles).all()
    tmp=cache.with_suffix('.partial.npz')
    np.savez_compressed(tmp,cls=embeddings,roles=roles,sig=SIGNATURE,doc_digest=digest)
    tmp.replace(cache)
    return embeddings.astype(np.float32),roles.astype(np.float32)

def flatten_cases(cases,use_role):
    feats=[]; targets=[]
    for case in cases:
        n=len(case['sentences'])
        position=np.asarray(case.get('original_indices',list(range(n))),dtype=np.float32)[:,None]/max(1,case.get('original_sentence_count',n)-1)
        mats=[case['cls'].astype(np.float32),position]
        if use_role:mats.append(case['roles'].astype(np.float32))
        feats.append(np.concatenate(mats,axis=1))
        if 'pseudo_y' not in case:
            raise ValueError('Cannot train on cases without training-only weak labels')
        targets.append(case['pseudo_y'].astype(np.float32))
    return torch.from_numpy(np.concatenate(feats)),torch.from_numpy(np.concatenate(targets))

class SentenceSelector(nn.Module):
    def __init__(self,use_roles=False):
        super().__init__()
        self.use_roles=use_roles
        d=768+1+(13 if use_roles else 0)
        self.net=nn.Sequential(nn.LayerNorm(d),nn.Linear(d,160),nn.ReLU(),
                               nn.Dropout(DROPOUT),nn.Linear(160,1))
    def forward(self,x):return self.net(x).squeeze(-1)

def case_selector_inputs(case,use_role):
    n=len(case['sentences'])
    pos=np.asarray(case.get('original_indices',list(range(n))),dtype=np.float32)[:,None]/max(1,case.get('original_sentence_count',n)-1)
    return torch.from_numpy(np.concatenate([case['cls'],pos]+([case['roles']] if use_role else []),axis=1))

SCORER=rouge_scorer.RougeScorer(['rouge1','rouge2','rougeL'],use_stemmer=True)

def sentence_tfidf_features(sentences):
    try:
        v=TfidfVectorizer(stop_words='english',ngram_range=(1,2),max_features=8000).fit_transform(sentences)
        cent=cosine_similarity(v,np.asarray(v.mean(axis=0)).ravel().reshape(1,-1)).ravel()
    except ValueError:
        v=None;cent=np.zeros(len(sentences),dtype=float)
    return v,cent

def budgeted_selection(sents,ranking,budget=SUMMARY_WORD_BUDGET,vectors=None):
    selected=[];words=0
    for idx in ranking:
        i=int(idx)
        wc=len(sents[i].split())
        if wc<5 or wc>budget or words+wc>budget:continue
        if vectors is not None and selected:
            if float(cosine_similarity(vectors[i],vectors[selected]).max())>=SUMMARY_REPETITION_THRESHOLD:
                continue
        selected.append(i);words+=wc
        if words>=budget:break
    return sorted(selected)

def create_extract(case,scores,word_budget=SUMMARY_WORD_BUDGET):
    rank=np.argsort(-scores,kind='stable')
    vectors,_=sentence_tfidf_features(case['sentences'])
    selected=budgeted_selection(case['sentences'],rank,budget=word_budget,vectors=vectors)
    return '\n'.join(case['sentences'][int(i)] for i in selected),selected

@torch.no_grad()
def validation_rougeL(model,use_roles):
    model.eval();values=[]
    for case in DEV_CASES:
        inputs=case_selector_inputs(case,use_roles).to(DEVICE)
        predictions=torch.sigmoid(model(inputs)).cpu().numpy()
        result,selected=create_extract(case,predictions)
        values.append(SCORER.score(case['reference'],result)['rougeL'].fmeasure)
    return float(np.mean(values))

def train_selector(label,use_roles):
    torch.manual_seed(SEED)
    net=SentenceSelector(use_roles).to(DEVICE)
    features,weak_y=flatten_cases(TRAIN_CASES,use_roles)
    positives=float(weak_y.sum().item());negatives=len(weak_y)-positives
    # Avoid pathological loss weights on short/noisy training samples.
    pos_weight=torch.tensor([min(6.0,max(1.0,negatives/max(1,positives)))],device=DEVICE)
    criterion=nn.BCEWithLogitsLoss(pos_weight=pos_weight)
    loader=DataLoader(TensorDataset(features,weak_y),batch_size=HEAD_BATCH_SIZE,
                      shuffle=True,generator=torch.Generator().manual_seed(SEED))
    optim=torch.optim.AdamW(net.parameters(),lr=HEAD_LR,weight_decay=0.01)
    dst=OUT_DIR/'model_checkpoints'/label
    dst.mkdir(parents=True,exist_ok=True)
    ck=dst/'best_selector.pt';best=-1;wait=0;history=[]
    for epoch in range(1,EPOCHS+1):
        net.train();running=0.;seen=0
        for x,y in loader:
            x=x.to(DEVICE);y=y.to(DEVICE)
            optim.zero_grad(set_to_none=True)
            logits=net(x)
            loss=criterion(logits,y)
            loss.backward()
            torch.nn.utils.clip_grad_norm_(net.parameters(),1.)
            optim.step()
            running+=float(loss.item())*len(y);seen+=len(y)
        metric=validation_rougeL(net,use_roles)
        history.append({'model':label,'epoch':epoch,'train_weak_bce':running/max(1,seen),
                        'val_rougeL':metric})
        print(f'{label}: epoch {epoch:02d} train_weak_bce={running/max(1,seen):.4f} val_rougeL={metric:.4f}')
        if metric>best+1e-5:
            best=metric;wait=0
            payload={'state_dict':{k:v.detach().cpu() for k,v in net.state_dict().items()},
                     'use_roles':use_roles,'epoch':epoch,'val_rougeL':metric,
                     'encoder_signature':SIGNATURE,'head_zip_digest':hashlib.sha256(Path(HEAD_ZIP).read_bytes()).hexdigest(),
                     'train_case_ids':[c['id'] for c in TRAIN_CASES],
                     'dev_case_ids':[c['id'] for c in DEV_CASES],
                     'seed':SEED,'budget':SUMMARY_WORD_BUDGET,'max_sents':MAX_SENTENCES_PER_DOC}
            tmp=ck.with_suffix('.temp');torch.save(payload,tmp);tmp.replace(ck)
        else:
            wait+=1
            if wait>=PATIENCE:break
    saved=torch.load(ck,map_location='cpu',weights_only=True)
    net.load_state_dict(saved['state_dict']);net.eval()
    return net,pd.DataFrame(history),{'model':label,'best_epoch':saved['epoch'],'val_rougeL':best,'checkpoint':str(ck)}

def lead_indices(sents,budget=SUMMARY_WORD_BUDGET):
    selected=[];count=0
    for i,s in enumerate(sents):
        n=len(s.split())
        if n<5:continue
        if count+n>budget:break
        selected.append(i);count+=n
    return selected

def tfidf_indices(sents):
    v,cent=sentence_tfidf_features(sents)
    return budgeted_selection(sents,np.argsort(-cent,kind='stable'),vectors=v)
