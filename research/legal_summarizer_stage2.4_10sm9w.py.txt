    def json_to_df(documents, split):
        rows, skipped, multiple_sets = [], [], []
        for doc_index, doc in enumerate(documents):
            annotations = doc.get('annotations') or []
            # A document may contain labels from several annotators. Never merge their
            # results: this would duplicate offsets and may invent contradictory labels.
            candidates = []
            for annotation_index, annotation in enumerate(annotations):
                if not isinstance(annotation, dict):
                    continue
                results = annotation.get('result') or []
                spans_here = [r['value'] for r in results if isinstance(r, dict)
                              and isinstance(r.get('value'), dict)
                              and 'text' in r['value']]
                if spans_here:
                    candidates.append((annotation_index, spans_here, annotation))

            if not candidates:
                skipped.append(doc_index)
                continue

            # Prefer an explicitly adjudicated/ground-truth annotation, if supplied.
            # The Hugging Face Parquet schema often contains only 'result' and lacks
            # adjudication metadata; in that case, use a documented deterministic policy.
            adjudicated = [item for item in candidates if
                           item[2].get('ground_truth') is True or
                           item[2].get('is_adjudicated') is True]
            if len(adjudicated) > 1:
                raise ValueError(f'{split} document {doc_index}: more than one annotation '
                                 'is marked adjudicated; review the source data.')
            selected_idx, spans, _ = (adjudicated[0] if adjudicated else candidates[0])
            selection_reason = 'adjudicated' if adjudicated else 'first_nonempty'

            if len(annotations) > 1:
                # Compare the entire labeled sentence sequence across nonempty sets.
                # Keep only counts/indices in audit, never duplicate the document text.
                def signature(annotator_spans):
                    records = [(v.get('start'), v.get('end'), v.get('text'),
                                tuple(v.get('labels') or ())) for v in annotator_spans]
                    return tuple(sorted(records, key=lambda x: (str(x[0]), str(x[1]), str(x[2]))))
                disagree = len({signature(item[1]) for item in candidates}) > 1
                multiple_sets.append(dict(doc_index=doc_index,
                                          total_annotation_sets=len(annotations),
                                          nonempty_annotation_sets=len(candidates),
                                          selected_annotation_index=selected_idx,
                                          selection_reason=selection_reason,
                                          annotation_sets_disagree=disagree))
            for value in spans:
                if not isinstance(value.get('start'), int) or not isinstance(value.get('end'), int):
                    raise ValueError(f'{split} document {doc_index}: missing sentence offsets')
                if value['end'] <= value['start'] or not value['text'].strip():
                    raise ValueError(f'{split} document {doc_index}: invalid or empty span')
                labels = value.get('labels') or []
                if len(labels) > 1:
                    raise ValueError('Expected one rhetorical role per sentence')
                label = labels[0] if labels else None
                if label is not None and label not in LABEL2ID:
                    raise ValueError(f'Unknown label: {label}')
                rows.append(dict(split=split, doc_index=doc_index, annotation_index=selected_idx, meta=doc.get('meta'),
                                 start=value['start'], end=value['end'], text=value['text'], label=label))
        df = pd.DataFrame(rows, columns=['split','doc_index','annotation_index','meta','start','end','text','label'])
        df = df.sort_values(['doc_index', 'start', 'end']).reset_index(drop=True)
        if df.duplicated(['doc_index', 'start', 'end']).any():
            raise ValueError(f'{split}: duplicate sentence spans')
        df['sentence_index'] = df.groupby('doc_index').cumcount()
        report = dict(raw_documents=len(documents), represented_documents=int(df.doc_index.nunique()),
                      sentences=len(df), skipped_document_indices=skipped,
                      unlabeled=int(df.label.isna().sum()),
                      multi_annotation_documents=len(multiple_sets),
                      multi_annotation_disagreements=sum(x['annotation_sets_disagree'] for x in multiple_sets),
                      annotation_selection_policy='adjudicated_if_marked_else_first_nonempty',
                      multi_annotation_details=multiple_sets)
        return df, report

    def make_neighbor_indices(frame):
        if frame.empty: return [],[]
        keys=frame[['doc_index','sentence_index']].reset_index(drop=True)
        assert keys.equals(keys.sort_values(['doc_index','sentence_index']).reset_index(drop=True))
        n=len(frame)
        prev=[i-1 if i>0 and keys.at[i-1,'doc_index']==keys.at[i,'doc_index'] else None for i in range(n)]
        nex=[i+1 if i+1<n and keys.at[i+1,'doc_index']==keys.at[i,'doc_index'] else None for i in range(n)]
        return prev,nex

    def pack_context_ids(target_ids,previous_ids,next_ids,max_length=256,target_cap=192):
        # Exact Stage 2.3 implementation. Target first segment; previous/next second segment.
        assert max_length>=8 and target_cap>=1
        tgt=list(target_ids[:min(target_cap,max_length-3)])
        budget=max_length-len(tgt)-3
        pmark=MARK_PREV if previous_ids else []
        nmark=MARK_NEXT if next_ids else []
        if len(pmark)+len(nmark)>budget: raise ValueError('Insufficient budget for neighbor markers')
        free=budget-len(pmark)-len(nmark)
        prev_cap=min(len(previous_ids),(free+1)//2) if previous_ids else 0
        next_cap=min(len(next_ids),free//2) if next_ids else 0
        leftover=free-prev_cap-next_cap
        prev_cap+=min(len(previous_ids)-prev_cap,leftover)
        leftover=free-prev_cap-next_cap
        next_cap+=min(len(next_ids)-next_cap,leftover)
        prev_part=list(previous_ids[-prev_cap:]) if prev_cap else []
        next_part=list(next_ids[:next_cap]) if next_cap else []
        context=pmark+prev_part+nmark+next_part
        ids=[TOKENIZER.cls_token_id]+tgt+[TOKENIZER.sep_token_id]+context+[TOKENIZER.sep_token_id]
        segment0=len(tgt)+2
        result={'input_ids':ids,'attention_mask':[1]*len(ids),
                'token_type_ids':[0]*segment0+[1]*(len(ids)-segment0)}
        assert len(ids)<=max_length
        assert ids[1:1+len(tgt)]==tgt
        return result

    class RecoveryRoleDataset(Dataset):
        def __init__(self,frame):
            texts=frame.text.astype(str).tolist()
            raw=TOKENIZER(texts,add_special_tokens=False,padding=False,
                          return_attention_mask=False)['input_ids']
            previous,following=make_neighbor_indices(frame)
            self.rows=[pack_context_ids(raw[i],
                        raw[previous[i]] if previous[i] is not None else [],
                        raw[following[i]] if following[i] is not None else [],
                        max_length=256,target_cap=192)
                       for i in range(len(texts))]
            self.labels=[LABEL2ID[x] for x in frame.label.tolist()]
        def __len__(self):return len(self.labels)
        def __getitem__(self,i):return {**self.rows[i],'labels':self.labels[i]}

def frame_digest(frame):
    h=hashlib.sha256()
    for doc,start,end,sentence,label in frame[['doc_index','start','end','text','label']].itertuples(index=False,name=None):
        h.update(json.dumps([int(doc),int(start),int(end),str(sentence),str(label)],
                            ensure_ascii=False).encode('utf-8'))
        h.update(b'\n')
    return h.hexdigest()

class JudgmentSequences(Dataset):
    def __init__(self,frame,embeddings):
        if len(frame)!=len(embeddings): raise ValueError('Feature row mismatch')
        self.frame=frame.reset_index(drop=True)
        self.embeddings=embeddings
        self.parts=[]
        for doc, part in self.frame.groupby('doc_index',sort=True):
            idx=part.index.to_numpy(dtype=np.int64)
            expected=np.arange(len(idx))
            actual=part.sentence_index.to_numpy()
            assert np.array_equal(actual,expected),f'Non-consecutive sentence order in document {doc}'
            self.parts.append(idx)
        assert sum(len(x) for x in self.parts)==len(frame)
    def __len__(self): return len(self.parts)
    def __getitem__(self,i):
        ix=self.parts[i]
        return (self.embeddings[ix].float(),
                torch.tensor([LABEL2ID[x] for x in self.frame.iloc[ix].label],dtype=torch.long),
                torch.tensor(ix,dtype=torch.long))

def collate_judgments(batch):
    b=len(batch);lengths=torch.tensor([len(x[0]) for x in batch],dtype=torch.long)
    maxlen=int(lengths.max());dim=batch[0][0].shape[-1]
    feats=torch.zeros(b,maxlen,dim,dtype=torch.float32)
    labels=torch.zeros(b,maxlen,dtype=torch.long)
    mask=torch.zeros(b,maxlen,dtype=torch.bool)
    orig_index=torch.full((b,maxlen),-1,dtype=torch.long)
    for j,(x,y,indices) in enumerate(batch):
        n=len(x);feats[j,:n]=x;labels[j,:n]=y;mask[j,:n]=True;orig_index[j,:n]=indices
    return dict(features=feats,labels=labels,mask=mask,lengths=lengths,indices=orig_index)

class SentenceBiLSTM(nn.Module):
    def __init__(self,in_dim=768,n_labels=13,hidden=128,dropout=0.30):
        super().__init__()
        self.norm=nn.LayerNorm(in_dim)
        self.drop=nn.Dropout(dropout)
        self.lstm=nn.LSTM(in_dim,hidden,num_layers=1,batch_first=True,bidirectional=True)
        self.classifier=nn.Linear(hidden*2,n_labels)
    def forward(self,x,lengths):
        x=self.drop(self.norm(x))
        packed=nn.utils.rnn.pack_padded_sequence(x,lengths.cpu(),batch_first=True,
                                                  enforce_sorted=False)
        packed_out,_=self.lstm(packed)
        output,_=nn.utils.rnn.pad_packed_sequence(packed_out,batch_first=True,total_length=x.shape[1])
        return self.classifier(self.drop(output))

def balanced_loss(emissions,gold,mask,mode,weights=class_weights,gamma=CFG['focal_gamma']):
    assert mask.dtype==torch.bool and mask.any()
    scores=emissions[mask]       # ONLY unpadded sentence logits
    target=gold[mask]
    log_probs=F.log_softmax(scores,dim=-1)
    true_log_prob=log_probs.gather(1,target[:,None]).squeeze(1)
    importance=weights[target]
    if mode=='weighted_ce':
        per_sentence=-true_log_prob*importance
    elif mode=='weighted_focal':
        p=true_log_prob.exp()
        per_sentence=-(1-p).pow(gamma)*true_log_prob*importance
    else:
        raise ValueError(f'Unsupported loss: {mode}')
    return per_sentence.sum()/importance.sum()

# Mathematical checks: focal gamma=0 equals weighted CE; pad gradients must be zero.
probe=torch.randn(2,4,len(LABELS),device=DEVICE,requires_grad=True)
labels_probe=torch.tensor([[0,1,2,3],[2,0,0,0]],device=DEVICE)
mask_probe=torch.tensor([[True,True,True,True],[True,False,False,False]],device=DEVICE)
a=balanced_loss(probe,labels_probe,mask_probe,'weighted_ce',class_weights)
b=balanced_loss(probe,labels_probe,mask_probe,'weighted_focal',class_weights,gamma=0)
assert torch.allclose(a,b,atol=1e-7)
a.backward()
assert probe.grad[~mask_probe].abs().sum()==0
assert torch.isfinite(a)
print('Balanced loss, focal gamma=0, and padding-gradient tests PASSED.')

def metrics_from(y,p):
    ids=np.arange(len(LABELS))
    return dict(accuracy=float(accuracy_score(y,p)),
                macro_f1=float(f1_score(y,p,labels=ids,average='macro',zero_division=0)),
                weighted_f1=float(f1_score(y,p,labels=ids,average='weighted',zero_division=0)))

@torch.no_grad()
def predict_docs(net,docs):
    net.eval()
    loader=DataLoader(docs,batch_size=CFG['doc_batch_size'],shuffle=False,
                      collate_fn=collate_judgments,num_workers=0)
    idxs=[];ys=[];preds=[]
    for batch in loader:
        x=batch['features'].to(DEVICE)
        logits=net(x,batch['lengths'])
        p=logits.argmax(-1).cpu()
        for i,n in enumerate(batch['lengths'].tolist()):
            idxs.extend(batch['indices'][i,:n].tolist())
            ys.extend(batch['labels'][i,:n].tolist())
            preds.extend(p[i,:n].tolist())
    ix=np.argsort(np.array(idxs))
    y=np.asarray(ys)[ix];p=np.asarray(preds)[ix]
    assert np.array_equal(np.asarray(idxs)[ix],np.arange(len(docs.frame)))
    assert np.array_equal(y,[LABEL2ID[v] for v in docs.frame.label])
    return y,p

def train_one(mode,seed=SEED):
    name=f'bilstm_{mode}_seed{seed}'
    folder=OUTPUT/name
    folder.mkdir(parents=True,exist_ok=True)
    conf=dict(name=name,mode=mode,seed=seed,cfg=CFG,
              class_weights=class_weights.cpu().tolist(),
              feature_signature=EXPECTED_ENCODER_SIGNATURE,
              train_digest=EXPECTED_TRAIN_DIGEST,dev_digest=EXPECTED_DEV_DIGEST)
    complete=folder/'training_complete.json'
    if (not FORCE_RETRAIN and complete.exists() and
        (folder/'head_config.json').exists() and (folder/'best_head.pt').exists() and
        (folder/'dev_metrics.json').exists() and
        json.loads((folder/'head_config.json').read_text())==conf):
        print('Reusing completed Stage 2.4 head:',name)
        return json.loads((folder/'dev_metrics.json').read_text())
    set_seed(seed)
    net=SentenceBiLSTM(hidden=CFG['hidden_size'],n_labels=len(LABELS),
                       dropout=CFG['dropout']).to(DEVICE)
    optimizer=torch.optim.AdamW(net.parameters(),lr=CFG['lr'],weight_decay=CFG['weight_decay'])
    best=-1.0; best_epoch=None; stale=0; history=[]
    print('\nTRAIN',name)
    for epoch in range(1,CFG['epochs']+1):
        net.train()
        loader=DataLoader(TRAIN_DOCS,batch_size=CFG['doc_batch_size'],shuffle=True,
                          collate_fn=collate_judgments,num_workers=0,
                          generator=torch.Generator().manual_seed(seed+epoch))
        total_loss=0.0;count_total=0
        for batch in tqdm(loader,desc=f'{name} epoch {epoch}',leave=False):
            x=batch['features'].to(DEVICE)
            gold=batch['labels'].to(DEVICE)
            mask=batch['mask'].to(DEVICE)
            emissions=net(x,batch['lengths'])
            loss=balanced_loss(emissions,gold,mask,mode)
            assert torch.isfinite(loss),f'{name}: nonfinite loss'
            optimizer.zero_grad(set_to_none=True)
            loss.backward()
            nn.utils.clip_grad_norm_(net.parameters(),CFG['max_grad_norm'])
            optimizer.step()
            n=int(mask.sum());total_loss+=float(loss.item())*n;count_total+=n
        y,p=predict_docs(net,DEV_DOCS)
        metric=metrics_from(y,p)
        history.append(dict(epoch=epoch,train_loss=total_loss/count_total,**metric))
        pd.DataFrame(history).to_csv(folder/'training_history.csv',index=False)
        print(name,'epoch',epoch,'loss',round(total_loss/count_total,4),
              'Macro F1',round(metric['macro_f1'],4),
              'Weighted F1',round(metric['weighted_f1'],4),
              'accuracy',round(metric['accuracy'],4))
        if metric['macro_f1']>best+1e-7:
            best=metric['macro_f1'];best_epoch=epoch;stale=0
            torch.save({'bilstm':net.state_dict(),'epoch':epoch,'config':conf},folder/'best_head.pt')
            (folder/'head_config.json').write_text(json.dumps(conf,indent=2))
            write_reports(folder,name,y,p,metric)
            (folder/'best_epoch.json').write_text(json.dumps(dict(epoch=epoch,**metric),indent=2))
        else:
            stale+=1
            if stale>=CFG['patience']:
                print('Early stopping — best epoch:',best_epoch)
                break
    (folder/'training_complete.json').write_text(json.dumps(
        {'finished':True,'epochs_run':len(history),'best_epoch':best_epoch},indent=2))
    return json.loads((folder/'dev_metrics.json').read_text())
