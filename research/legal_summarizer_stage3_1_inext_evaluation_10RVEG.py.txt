def checkpoint_signature(path):
    h=hashlib.sha256()
    for name in ('config.json','tokenizer.json','model.safetensors'):
        p=Path(path)/name
        if not p.is_file(): raise FileNotFoundError(p)
        h.update(name.encode());h.update(str(p.stat().st_size).encode())
        with p.open('rb') as f:
            h.update(f.read(128*1024))
            if p.stat().st_size>128*1024:
                f.seek(-128*1024,2);h.update(f.read(128*1024))
    return h.hexdigest()

class SentenceBiLSTM(nn.Module):
    # Exactly the same architecture as saved Stage 2.3 Softmax head
    def __init__(self,in_dim=768,n_labels=13,hidden=128,dropout=0.3):
        super().__init__()
        self.norm=nn.LayerNorm(in_dim)
        self.drop=nn.Dropout(dropout)
        self.lstm=nn.LSTM(in_dim,hidden,num_layers=1,batch_first=True,bidirectional=True)
        self.classifier=nn.Linear(hidden*2,n_labels)
    def forward(self,x,lengths):
        x=self.drop(self.norm(x))
        packed=nn.utils.rnn.pack_padded_sequence(x,lengths.cpu(),batch_first=True,enforce_sorted=False)
        packed_out,_=self.lstm(packed)
        output,_=nn.utils.rnn.pad_packed_sequence(packed_out,batch_first=True,total_length=x.shape[1])
        return self.classifier(self.drop(output))

def pack_context_ids(target_ids,previous_ids,next_ids,max_length=256,target_cap=192):
    tgt=list(target_ids[:min(target_cap,max_length-3)])
    budget=max_length-len(tgt)-3
    pmark=MARK_PREV if previous_ids else []
    nmark=MARK_NEXT if next_ids else []
    free=budget-len(pmark)-len(nmark)
    if free<0: raise ValueError('No token budget for context markers')
    prev_cap=min(len(previous_ids),(free+1)//2) if previous_ids else 0
    next_cap=min(len(next_ids),free//2) if next_ids else 0
    left=free-prev_cap-next_cap
    prev_cap+=min(len(previous_ids)-prev_cap,left)
    left=free-prev_cap-next_cap
    next_cap+=min(len(next_ids)-next_cap,left)
    prev_part=list(previous_ids[-prev_cap:]) if prev_cap else []
    next_part=list(next_ids[:next_cap]) if next_cap else []
    context=pmark+prev_part+nmark+next_part
    ids=[TOKENIZER.cls_token_id]+tgt+[TOKENIZER.sep_token_id]+context+[TOKENIZER.sep_token_id]
    segment0=len(tgt)+2
    item={'input_ids':ids,'attention_mask':[1]*len(ids),
          'token_type_ids':[0]*segment0+[1]*(len(ids)-segment0)}
    assert len(ids)<=max_length
    return item

def canonical_rel_name(path):
    # Parse zip path, locate IN-Ext anywhere in nesting; case-insensitive.
    parts=path.replace('\\','/').split('/')
    hits=[i for i,p in enumerate(parts) if p.lower().replace('_','-') in ('in-ext','inext')]
    if not hits:return None
    return parts[hits[-1]+1:]

def build_inext_index(z):
    result={'judgement':{},'A1':{},'A2':{}}
    for info in z.infolist():
        if info.is_dir() or info.file_size==0 or info.file_size>20_000_000:continue
        rel=canonical_rel_name(info.filename)
        if not rel or len(rel)<2:continue
        rel_low=[p.lower() for p in rel]
        if rel_low[0] in ('judgement','judgment'):
            # If judgments are nested, preserve suffix of filename to avoid collisions.
            key='/'.join(rel[1:]).lower()
            result['judgement'][key]=info.filename
        elif (len(rel)>=4 and rel_low[0]=='summary' and rel_low[1]=='full'
              and rel_low[2] in ('a1','a2')):
            key='/'.join(rel[3:]).lower()
            result[rel[2].upper()][key]=info.filename
    pairs=[]
    for k in sorted(set(result['judgement'])&set(result['A1'])&set(result['A2'])):
        if k.endswith(('.txt','.text')):
            pairs.append({'case_id':k,'judgment':result['judgement'][k],
                          'a1':result['A1'][k],'a2':result['A2'][k]})
    return pairs,result

def normalize_for_segmentation(text):
    text=text.replace('\u00ad','')
    text=re.sub(r'(?<=\w)-\s*\n\s*(?=\w)','',text)
    return re.sub(r'\s+',' ',text).strip()

def segment_document(text):
    cleaned=normalize_for_segmentation(text)
    NLP.max_length=max(NLP.max_length,len(cleaned)+100)
    sentences=[s.text.strip() for s in NLP(cleaned).sents if re.search(r'[A-Za-z]',s.text)]
    if len(sentences)<4:raise ValueError(f'Too few sentences found: {len(sentences)}')
    if len(sentences)>MAX_SENTENCES_PER_CASE:
        raise ValueError(f'Judgment contains {len(sentences)} sentences; hard limit {MAX_SENTENCES_PER_CASE}. '
                         'Increase MAX_SENTENCES_PER_CASE after checking RAM; nothing has been truncated.')
    return sentences

@torch.inference_mode()
def predict_roles(sentences,batch_size=8):
    encoded_tokens=TOKENIZER(sentences,add_special_tokens=False,padding=False,
                             return_attention_mask=False)['input_ids']
    packed=[]
    for i,ids in enumerate(encoded_tokens):
        before=encoded_tokens[i-1] if i else []
        after=encoded_tokens[i+1] if i+1<len(encoded_tokens) else []
        packed.append(pack_context_ids(ids,before,after,
                                      max_length=CFG['max_length'],
                                      target_cap=CFG['max_target_context']))
    all_features=[]
    from tqdm.auto import tqdm
    for start in tqdm(range(0,len(packed),batch_size),desc='Encode sentences'):
        batch=COLLATOR(packed[start:start+batch_size])
        inputs={k:v.to(DEVICE) for k,v in batch.items()}
        with torch.amp.autocast(device_type='cuda',enabled=(DEVICE.type=='cuda')):
            cls=ENCODER.base_model(**inputs,return_dict=True).last_hidden_state[:,0,:]
        # Original Stage 2.3 encoder saved float16 embeddings before loading the BiLSTM.
        all_features.append(cls.float().cpu().to(torch.float16))
    features=torch.cat(all_features,dim=0).float().unsqueeze(0).to(DEVICE)
    lengths=torch.tensor([len(sentences)],dtype=torch.long)
    logits=HEAD(features,lengths)[0].float()
    probs=torch.softmax(logits,dim=-1).cpu().numpy()
    role_idx=probs.argmax(axis=1)
    return role_idx,probs

# Fixed rules, chosen before seeing IN-Ext results.
ROLE_GROUPS = {
    'Facts':['FAC'], 'Issues':['ISSUE'],
    'Petitioner arguments':['ARG_PETITIONER'], 'Respondent arguments':['ARG_RESPONDENT'],
    'Legal reasoning':['ANALYSIS','RATIO','PRE_RELIED','STA'],
    'Decision':['RPC'],
}
ROLE_QUOTAS = {'Facts':3,'Issues':1,'Petitioner arguments':1,'Respondent arguments':1,
               'Legal reasoning':4,'Decision':2}
ROLE_PRIORITY=['Decision','Issues','Facts','Legal reasoning','Petitioner arguments','Respondent arguments']

# Sentence importance is computed without gold summaries.
def sentence_features(sentences):
    v=TfidfVectorizer(stop_words='english',ngram_range=(1,2),max_features=8000).fit_transform(sentences)
    cent=np.asarray(v.mean(axis=0)).ravel()[None,:]
    vals=cosine_similarity(v,cent).ravel()
    vals/=max(float(vals.max()),1e-8)
    return v,vals

def select_indices(sentences,vectors,ranking,budget=SUMMARY_WORD_BUDGET,chosen=None,
                   redundancy=REDUNDANCY_COSINE):
    selected=list(chosen or [])
    total=sum(len(sentences[i].split()) for i in selected)
    for idx in ranking:
        i=int(idx)
        if i in selected:continue
        wc=len(sentences[i].split())
        if wc<5 or total+wc>budget:continue
        if selected:
            similarities=cosine_similarity(vectors[i],vectors[selected]).ravel()
            if len(similarities) and float(max(similarities))>=redundancy:continue
        selected.append(i);total+=wc
        if total>=budget:break
    return selected

def make_lead_summary(sentences,budget=SUMMARY_WORD_BUDGET):
    # Lead takes original sentences; no skipping to optimize relevance.
    chosen=[];words=0
    for i,s in enumerate(sentences):
        wc=len(s.split())
        if wc<5:continue
        if words+wc>budget:break
        chosen.append(i);words+=wc
    return chosen

def make_tfidf_summary(sentences,vectors,cent,budget=SUMMARY_WORD_BUDGET):
    rank=np.argsort(-cent,kind='stable')
    return select_indices(sentences,vectors,rank,budget)

def make_role_summary(sentences,vectors,cent,role_probs,budget=SUMMARY_WORD_BUDGET):
    # Reserve slots across important roles using SOFT probs to tolerate missed ISSUE hard labels.
    # Use only predictions + text, not expert summaries or document gold annotations.
    per_section={key:role_probs[:,[LABELS.index(r) for r in labels]].sum(axis=1)
                 for key,labels in ROLE_GROUPS.items()}
    chosen=[]
    for section in ROLE_PRIORITY:
        mass=per_section[section]
        # Probabilities are conditional uncalibrated scores, treated only as *ranking*.
        score=.70*mass + .30*cent
        ranking=np.argsort(-score,kind='stable')
        picks_for_section=0
        for idx in ranking:
            before=len(chosen)
            chosen=select_indices(sentences,vectors,[idx],budget,chosen)
            if len(chosen)>before:picks_for_section+=1
            if picks_for_section>=ROLE_QUOTAS[section]:break
    # Fill any spare budget with central, nonredundant sentences (same scorer as baseline).
    chosen=select_indices(sentences,vectors,np.argsort(-cent,kind='stable'),budget,chosen)
    return chosen

def summary_as_text(sentences,indices):
    return '\n'.join(sentences[i] for i in sorted(indices))

        tf_ix=make_tfidf_summary(sents,vectors,cent)
        role_ids,prob=predict_roles(sents,batch_size=EMBEDDING_BATCH_SIZE)
        role_ix=make_role_summary(sents,vectors,cent,prob)
        indices={'Lead':lead_ix,'TFIDF':tf_ix,'Role-guided':role_ix}
        for method,ixs in indices.items():
            generated=summary_as_text(sents,ixs)
            wc=len(generated.split())
            assert wc<=SUMMARY_WORD_BUDGET
            assert all(0<=k<len(sents) for k in ixs)
            for annotator in ('A1','A2'):
                reference=case[annotator]
                scores=scorer.score(reference,generated)
                report.append({'case_id':cid,'reference':annotator,'method':method,
                               'input_sentences':len(sents),'input_words':len(case['judgment'].split()),
                               'summary_sentences':len(ixs),'summary_words':wc,
                               'compression_ratio':round(wc/max(1,len(case['judgment'].split())),4),
                               'rouge1_f1':scores['rouge1'].fmeasure,
                               'rouge2_f1':scores['rouge2'].fmeasure,
                               'rougeL_f1':scores['rougeL'].fmeasure})
            audit.append({'case_id':cid,'method':method,'indices':sorted(int(i) for i in ixs),
                          'source_sentences':[{'sid':f'S{i+1:04d}','text':sents[i],
                                               'predicted_role':LABELS[int(role_ids[i])]}
                                              for i in sorted(ixs)]})
            if len(examples)<3:
                examples.setdefault(cid,{})[method]=generated
        print('Completed:',cid,'|',len(sents),'sentences')
    except Exception as e:
        errors.append({'case_id':cid,'type':type(e).__name__,'error':str(e)})
        print('Case skipped, not silently truncated:',cid,type(e).__name__,str(e)[:220])

if not report:
    raise RuntimeError('No cases successfully evaluated. See errors above. Do NOT report ROUGE values.')

results=pd.DataFrame(report)
