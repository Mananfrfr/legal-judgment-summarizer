def make_neighbor_indices(frame):
    """Exactly one previous/next from the SAME doc, ordered by sentence_index."""
    if frame.empty: return [],[]
    keys=frame[['doc_index','sentence_index']].reset_index(drop=True)
    # Preprocessor must already sort sentences within each document.
    assert keys.equals(keys.sort_values(['doc_index','sentence_index']).reset_index(drop=True))
    n=len(frame)
    prev=[i-1 if i>0 and keys.at[i-1,'doc_index']==keys.at[i,'doc_index'] else None for i in range(n)]
    nex=[i+1 if i+1<n and keys.at[i+1,'doc_index']==keys.at[i,'doc_index'] else None for i in range(n)]
    return prev,nex

def pack_context_ids(target_ids, previous_ids, next_ids, max_length=256, target_cap=192):
    """First BERT segment is the target; second consists of balanced neighboring text."""
    assert max_length>=8 and target_cap>=1
    tgt=list(target_ids[:min(target_cap,max_length-3)])
    budget=max_length-len(tgt)-3  # CLS, first SEP, final SEP
    pmark=MARK_PREV if previous_ids else []
    nmark=MARK_NEXT if next_ids else []
    if len(pmark)+len(nmark)>budget:
        raise ValueError('Not enough context-token budget for markers')
    free=budget-len(pmark)-len(nmark)
    prev_cap=min(len(previous_ids), (free+1)//2) if previous_ids else 0
    next_cap=min(len(next_ids), free//2) if next_ids else 0
    # If one neighbor is short/missing, let the other use the unused space.
    leftover=free-prev_cap-next_cap
    prev_cap+=min(len(previous_ids)-prev_cap,leftover)
    leftover=free-prev_cap-next_cap
    next_cap+=min(len(next_ids)-next_cap,leftover)
    prev_part=list(previous_ids[-prev_cap:]) if prev_cap else []  # closest end of previous
    next_part=list(next_ids[:next_cap]) if next_cap else []        # closest start of next
    context=pmark+prev_part+nmark+next_part
    ids=[TOKENIZER.cls_token_id]+tgt+[TOKENIZER.sep_token_id]+context+[TOKENIZER.sep_token_id]
    segment0=len(tgt)+2
    result={'input_ids':ids,'attention_mask':[1]*len(ids),
            'token_type_ids':[0]*segment0+[1]*(len(ids)-segment0)}
    assert len(ids)<=max_length
    assert ids[1:1+len(tgt)]==tgt, 'The target must remain unchanged after truncation'
    return result

class RoleDataset(Dataset):
    def __init__(self, frame, mode, tokenizer=TOKENIZER):
        if mode not in ('target256','context256'): raise ValueError(mode)
        self.mode=mode
        texts=frame.text.astype(str).tolist()
        self.labels=[LABEL2ID[x] for x in frame.label.tolist()]
        if mode=='target256':
            batch=tokenizer(texts,truncation=True,max_length=CFG['max_length'],
                            add_special_tokens=True,padding=False,return_attention_mask=True)
            self.rows=[{key:value[i] for key,value in batch.items()} for i in range(len(texts))]
        else:
            raw_tokens=tokenizer(texts,add_special_tokens=False,padding=False,return_attention_mask=False)['input_ids']
            previous,following=make_neighbor_indices(frame)
            self.rows=[pack_context_ids(raw_tokens[i],
                                        raw_tokens[previous[i]] if previous[i] is not None else [],
                                        raw_tokens[following[i]] if following[i] is not None else [],
                                        max_length=CFG['max_length'],target_cap=CFG['max_target_context'])
                       for i in range(len(texts))]
        assert len(self.rows)==len(self.labels)
    def __len__(self): return len(self.labels)
    def __getitem__(self,i): return {**self.rows[i],'labels':self.labels[i]}

@torch.no_grad()
def predict_dev(model,loader):
    model.eval(); true_ids=[];pred_ids=[]
    for batch in tqdm(loader,desc='Validation',leave=False):
        gold=batch['labels']
        inputs={k:v.to(DEVICE) for k,v in batch.items() if k!='labels'}
        with torch.amp.autocast('cuda',enabled=(DEVICE.type=='cuda')):
            logits=model(**inputs).logits
        true_ids.extend(gold.tolist());pred_ids.extend(logits.argmax(dim=-1).cpu().tolist())
    return np.array(true_ids),np.array(pred_ids)

def score(y,p):
    classes=list(range(len(LABELS)))
    return dict(accuracy=float(accuracy_score(y,p)),
                macro_f1=float(f1_score(y,p,labels=classes,average='macro',zero_division=0)),
                weighted_f1=float(f1_score(y,p,labels=classes,average='weighted',zero_division=0)))

def make_loader(dataset,batch_size,shuffle=False):
    generator=torch.Generator().manual_seed(SEED)
    return DataLoader(dataset,batch_size=batch_size,shuffle=shuffle,
                      collate_fn=COLLATOR,generator=generator)

def run_experiment(mode):

        print('SKIPPED:',mode,'dataset is not loaded');return None
    if not RUN_TRAINING:
        print('SKIPPED:',mode,'RUN_TRAINING=False');return None
    if DEVICE.type!='cuda':
        print('SKIPPED:',mode,'select Colab T4 GPU');return None
    folder=OUTPUT/mode
    folder.mkdir(exist_ok=True,parents=True)
    print('\n'+'='*60+'\nEXPERIMENT:',mode,'\nOutput folder:',folder)
    train_ds,dev_ds=DATASETS[mode]
    train_loader=make_loader(train_ds,CFG['train_batch_size'],shuffle=True)
    dev_loader=make_loader(dev_ds,CFG['eval_batch_size'])
    set_seed(SEED)
    model=AutoModelForSequenceClassification.from_pretrained(
        MODEL_ID,num_labels=len(LABELS),id2label=ID2LABEL,label2id=LABEL2ID).to(DEVICE)
    opt=torch.optim.AdamW(model.parameters(),lr=CFG['learning_rate'],weight_decay=CFG['weight_decay'])
    steps_per_epoch=math.ceil(len(train_loader)/CFG['grad_accum_steps'])
    steps_total=steps_per_epoch*CFG['epochs']
    scheduler=get_linear_schedule_with_warmup(opt,
        num_warmup_steps=int(steps_total*CFG['warmup_ratio']),
        num_training_steps=steps_total)
    scaler=torch.amp.GradScaler('cuda',enabled=True)
    best=-1;history=[]
    opt.zero_grad(set_to_none=True)
    for epoch in range(1,CFG['epochs']+1):
        model.train();losses=[]
        batches=len(train_loader)
        progress=tqdm(enumerate(train_loader),total=batches,desc=f'{mode} epoch {epoch}/{CFG["epochs"]}')
        for step,batch in progress:
            inputs={k:v.to(DEVICE) for k,v in batch.items()}
            group_start=(step//CFG['grad_accum_steps'])*CFG['grad_accum_steps']
            group_size=min(CFG['grad_accum_steps'],batches-group_start)
            with torch.amp.autocast('cuda'):
                loss=model(**inputs).loss
            if not torch.isfinite(loss).item(): raise RuntimeError('Nonfinite loss in '+mode)
            scaler.scale(loss/group_size).backward()
            losses.append(float(loss.item()))
            if (step+1)%CFG['grad_accum_steps']==0 or (step+1)==batches:
                scaler.unscale_(opt)
                torch.nn.utils.clip_grad_norm_(model.parameters(),CFG['max_grad_norm'])
                before=scaler.get_scale()
                scaler.step(opt)
                scaler.update()
                after=scaler.get_scale()
                if after>=before: scheduler.step()  # Do not advance after an AMP-overflow skip.
                opt.zero_grad(set_to_none=True)
            if (step+1)%20==0: progress.set_postfix(loss=f'{np.mean(losses[-20:]):.3f}')
        y,p=predict_dev(model,dev_loader)
        metrics=score(y,p)
        history.append({'epoch':epoch,'training_loss':float(np.mean(losses)),**metrics})
        pd.DataFrame(history).to_csv(folder/'training_history.csv',index=False)
        print('Epoch',epoch,metrics)
        if metrics['macro_f1']>best:
            best=metrics['macro_f1']
            print('  New best Macro F1: checkpoint saved for',mode)
            (folder/'best_epoch.json').write_text(json.dumps({'epoch':epoch,**metrics},indent=2))
            checkpoint=folder/'best_checkpoint'
            model.save_pretrained(checkpoint,safe_serialization=True)
            TOKENIZER.save_pretrained(checkpoint)
            save_dev_reports(mode,y,p,metrics)
    del model,opt,scheduler
    torch.cuda.empty_cache()
    return json.loads((folder/'dev_metrics.json').read_text())
