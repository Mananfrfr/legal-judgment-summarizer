def _head_config_looks_right(data):
    try:
        cfg=json.loads(data)
        return isinstance(cfg.get('feature_checkpoint_signature'),str) and len(cfg['feature_checkpoint_signature'])==64
    except (ValueError,TypeError,UnicodeDecodeError,AttributeError):
        return False

def _normalized_head_zip_from_archive(path):
    """Find a real trained head in a ZIP, even when wrapped in an extra folder."""
    try:
        with zipfile.ZipFile(path) as z:
            members={x.filename.replace('\\','/').lstrip('./'):x.filename
                     for x in z.infolist() if not x.is_dir()}
            for name in sorted(members):
                if not name.endswith('head_config.json'):continue
                parent=name.rsplit('/',1)[0] if '/' in name else ''
                best=f'{parent}/best_head.pt' if parent else 'best_head.pt'
                if best not in members:continue
                if not _head_config_looks_right(z.read(members[name])):continue
                # Already in expected layout; avoid rewriting its bytes or hash.
                if name=='bilstm_softmax/head_config.json' and best=='bilstm_softmax/best_head.pt':
                    if z.testzip() is None:
                        return path
                    continue
                output=ASSET_STAGING/('normalized_head_'+hashlib.sha256(str(path).encode()).hexdigest()[:12]+'.zip')
                with zipfile.ZipFile(output,'w',compression=zipfile.ZIP_DEFLATED) as normalized:
                    normalized.writestr('bilstm_softmax/head_config.json',z.read(members[name]))
                    normalized.writestr('bilstm_softmax/best_head.pt',z.read(members[best]))
                print('Recognized ZIP with nested head files:',path)
                print('Normalized trained head bundle:',output)
                return output
    except (OSError,ValueError,zipfile.BadZipFile,RuntimeError) as exc:
        print('Cannot use candidate ZIP:',path,'(',type(exc).__name__,')')
    return None

def _normalized_head_zip_from_folder(folder):
    cfg=folder/'head_config.json'
    pt=folder/'best_head.pt'
    if not (cfg.is_file() and pt.is_file() and _head_config_looks_right(cfg.read_bytes())):
        return None
    output=ASSET_STAGING/('normalized_head_folder_'+hashlib.sha256(str(folder).encode()).hexdigest()[:12]+'.zip')
    with zipfile.ZipFile(output,'w',compression=zipfile.ZIP_DEFLATED) as z:
        z.write(cfg,arcname='bilstm_softmax/head_config.json')
        z.write(pt,arcname='bilstm_softmax/best_head.pt')
    print('Recognized extracted Stage 2.3 head folder:',folder)
    print('Repacked for compatibility:',output)
    return output

def _candidate_encoder_signature(path):
    h=hashlib.sha256()
    for name in ('config.json','tokenizer.json','model.safetensors'):
        p=path/name
        if not p.is_file():return None
        h.update(name.encode());h.update(str(p.stat().st_size).encode())
        with p.open('rb') as f:
            h.update(f.read(128*1024))
            if p.stat().st_size>128*1024:
                f.seek(-128*1024,2);h.update(f.read(128*1024))
    return h.hexdigest()

class SentenceBiLSTM(nn.Module):
    # Exactly the same architecture as saved Stage 2.3 Softmax head
    def __init__(self,in_dim=768,n_labels=13,hidden=128,dropout=0.3):
        super().__init__()
        self.norm=nn.LayerNorm(in_dim)
        self.drop=nn.Dropout(dropout)
        self.lstm=nn.LSTM(in_dim,hidden,num_layers=1,batch_first=True,bidirectional=True)
        self.classifier=nn.Linear(hidden*2,n_labels)
    def forward(self,x,lengths):
        x=self.drop(self.norm(x))
        packed=nn.utils.rnn.pack_padded_sequence(x,lengths.cpu(),batch_first=True,enforce_sorted=False)
        packed_out,_=self.lstm(packed)
        output,_=nn.utils.rnn.pad_packed_sequence(packed_out,batch_first=True,total_length=x.shape[1])
        return self.classifier(self.drop(output))

def pack_context_ids(target_ids,previous_ids,next_ids,max_length=256,target_cap=192):
    tgt=list(target_ids[:min(target_cap,max_length-3)])
    budget=max_length-len(tgt)-3
    pmark=MARK_PREV if previous_ids else []
    nmark=MARK_NEXT if next_ids else []
    free=budget-len(pmark)-len(nmark)
    if free<0: raise ValueError('No token budget for context markers')
    prev_cap=min(len(previous_ids),(free+1)//2) if previous_ids else 0
    next_cap=min(len(next_ids),free//2) if next_ids else 0
    left=free-prev_cap-next_cap
    prev_cap+=min(len(previous_ids)-prev_cap,left)
    left=free-prev_cap-next_cap
    next_cap+=min(len(next_ids)-next_cap,left)
    prev_part=list(previous_ids[-prev_cap:]) if prev_cap else []
    next_part=list(next_ids[:next_cap]) if next_cap else []
    context=pmark+prev_part+nmark+next_part
    ids=[TOKENIZER.cls_token_id]+tgt+[TOKENIZER.sep_token_id]+context+[TOKENIZER.sep_token_id]
    segment0=len(tgt)+2
    item={'input_ids':ids,'attention_mask':[1]*len(ids),
          'token_type_ids':[0]*segment0+[1]*(len(ids)-segment0)}
    assert len(ids)<=max_length
    return item

def md5sum(path):
    h=hashlib.md5()
    with open(path,'rb') as f:
        for chunk in iter(lambda:f.read(4*1024*1024),b''):h.update(chunk)
    return h.hexdigest()

def dataset_tail(path, dataset_name):
    parts=path.replace('\\','/').split('/')
    aliases={dataset_name.lower(), dataset_name.lower().replace('-','')}
    hits=[i for i,p in enumerate(parts) if p.lower().replace('_','-') in aliases]
    return parts[hits[-1]+1:] if hits else None

def index_official_dataset(z):
    inabs={'train':{'judgement':{},'summary':{}},'test':{'judgement':{},'summary':{}}}
    inext={}
    for info in z.infolist():
        if info.is_dir() or not info.file_size or info.file_size>20_000_000: continue
        rel=dataset_tail(info.filename,'IN-Abs')
        if rel and len(rel)>=3:
            low=[x.lower() for x in rel]
            split='train' if low[0]=='train-data' else 'test' if low[0]=='test-data' else None
            typ=low[1]
            if split and typ in ('judgement','judgment','summary'):
                typ='judgement' if typ=='judgment' else typ
                key='/'.join(x.lower() for x in rel[2:])
                if key.endswith('.txt'):
                    inabs[split][typ][key]=info.filename
        rel=dataset_tail(info.filename,'IN-Ext')
        if rel and len(rel)>=2 and rel[0].lower() in ('judgement','judgment'):
            key='/'.join(x.lower() for x in rel[1:])
            if key.endswith('.txt'): inext[key]=info.filename
    pairs={}
    for split in ('train','test'):
        d=inabs[split]
        keys=sorted(set(d['judgement']) & set(d['summary']))
        pairs[split]=[{'id':key,'judgment_path':d['judgement'][key],
                       'summary_path':d['summary'][key], 'split':split} for key in keys]
    return pairs, inext, inabs

def clean_text(raw):
    return raw.decode('utf-8',errors='replace') if isinstance(raw,bytes) else str(raw)

def normalized_doc(s):
    return re.sub(r'[^a-z0-9]+',' ',s.lower()).strip()

def document_signatures(s):
    s=normalized_doc(s)
    h=lambda x: hashlib.sha256(x.encode('utf-8')).hexdigest()
    return (h(s),h(s[:4000]),len(s))

def normalize_for_segmentation(text):
    text=text.replace('\u00ad','')
    text=re.sub(r'(?<=\w)-\s*\n\s*(?=\w)','',text)
    return re.sub(r'\s+',' ',text).strip()

def split_sentences(text):
    clean=normalize_for_segmentation(text)
    NLP.max_length=max(NLP.max_length,len(clean)+100)
    return [s.text.strip() for s in NLP(clean).sents if re.search('[A-Za-z]',s.text)]

def select_window_indices(n,limit):
    if n<=limit:return list(range(n))
    first=round(limit*0.5);last=round(limit*0.25);middle=limit-first-last
    mid_start=max(first,(n-middle)//2)
    idx=list(range(first))+list(range(mid_start,mid_start+middle))+list(range(n-last,n))
    assert len(idx)==len(set(idx))==limit
    return sorted(idx)

def unigram_bigram_counts(text):
    from collections import Counter
    toks=ROUGE_TOKENIZER.tokenize(text)
    return Counter(toks),Counter(zip(toks,toks[1:]))

def tfidf_weak_labels(sentences,reference,positive_frac=POSITIVE_FRACTION):
    """Stage 3.4's same positive-fraction sentence-ranking rule."""
    assert sentences
    tfidf=TfidfVectorizer(stop_words='english',ngram_range=(1,2),max_features=12000,
                          token_pattern=r'(?u)\b\w\w+\b')
    try:
        vectors=tfidf.fit_transform(list(sentences)+[reference])
        similarity=cosine_similarity(vectors[:-1],vectors[-1]).ravel()
    except ValueError:
        similarity=np.zeros(len(sentences),dtype=np.float32)
    eligible=[i for i,s in enumerate(sentences) if len(s.split())>=5]
    npos=min(len(eligible),max(1,round(len(eligible)*positive_frac)))
    rank=sorted(eligible,key=lambda i:(-similarity[i],i))
    y=np.zeros(len(sentences),dtype=np.float32)
    y[rank[:npos]]=1
    return y

    """Improvement in clipped reference n-gram recall, counting repeats correctly."""
    return sum(min(count,max(0,reference_counts[gram]-used[gram]))
               for gram,count in candidate.items())

def greedy_oracle_labels(sentences,reference,budget=SUMMARY_WORD_BUDGET):
    """Training-only greedy ROUGE-1/2 *marginal coverage*, with word-budget and cost.
    This is NOT a ROUGE-L oracle and NOT a supervised human sentence annotation.
    It does not consume the dev labels unless called explicitly as a diagnostic.
    """
    from collections import Counter
    ref1,ref2=unigram_bigram_counts(reference)
    candidate_ngrams=[unigram_bigram_counts(s) for s in sentences]
    used1,used2=Counter(),Counter()
    selected=[];words_used=0;remaining=set(range(len(sentences)))
    while remaining:
        best_i=None;best_utility=0.0
        for i in sorted(remaining):
            wc=len(sentences[i].split())
            if wc<5 or wc>budget-words_used: continue
            c1,c2=candidate_ngrams[i]
            gain1=clipped_overlap_gain(used1,c1,ref1)/max(1,sum(ref1.values()))
            gain2=clipped_overlap_gain(used2,c2,ref2)/max(1,sum(ref2.values()))
            # Prioritize previously uncovered reference bigrams, with mild penalty for long sentences.
            utility=(0.4*gain1+0.6*gain2)/(wc**0.35)
            if utility>best_utility+1e-12:
                best_utility=utility;best_i=i
        if best_i is None: break  # No sentence adds relevant n-gram coverage within remaining budget.
        selected.append(best_i)
        words_used+=len(sentences[best_i].split())
        remaining.remove(best_i)
        c1,c2=candidate_ngrams[best_i]
        used1.update(c1);used2.update(c2)
    labels=np.zeros(len(sentences),dtype=np.float32)
    labels[selected]=1.0
    # If a human summary has no lexical match, don't silently create an all-negative training case.
    if not selected and sentences:
        fallback=tfidf_weak_labels(sentences,reference)
        if fallback.any():labels[int(np.flatnonzero(fallback)[0])]=1.0
    return labels

def tfidf_budgeted_labels(sentences,reference,budget=SUMMARY_WORD_BUDGET):
    """Budget-matched TF-IDF extraction teacher (training human summaries ONLY).
    Same 400-word constraint as greedy ROUGE; selects sentence importance labels.
    The reference is used to construct TRAIN labels, never as a model input.
    """
    assert sentences and budget>0
    try:
        vec=TfidfVectorizer(stop_words='english',ngram_range=(1,2),max_features=12000,
            token_pattern=r'(?u)\b\w\w+\b').fit_transform(list(sentences)+[reference])
        similarities=cosine_similarity(vec[:-1],vec[-1]).ravel()
    except ValueError:
        similarities=np.zeros(len(sentences),dtype=np.float32)
    ranking=sorted(range(len(sentences)),key=lambda i:(-float(similarities[i]),i))
    y=np.zeros(len(sentences),dtype=np.float32)
    used=0
    for i in ranking:
        wc=len(sentences[i].split())
        if wc<5 or wc>budget-used:continue
        y[i]=1.;used+=wc
        if used==budget:break
    return y

def lexical_reference_coverage(sentences,reference):
    from collections import Counter
    r1,r2=unigram_bigram_counts(reference)
    union1,union2=Counter(),Counter()
    for s in sentences:
        a,b=unigram_bigram_counts(s)
        # Source candidates' aggregate n-gram availability (no word budget).
        union1.update(a)
        union2.update(b)
    cover1=sum(min(n,union1[t]) for t,n in r1.items())/max(1,sum(r1.values()))
    cover2=sum(min(n,union2[t]) for t,n in r2.items())/max(1,sum(r2.values()))
    return float(cover1),float(cover2)

def effective_neighbors(indices,j):
    left=indices[j-1] if j and indices[j-1]+1==indices[j] else None
    right=indices[j+1] if j+1<len(indices) and indices[j]+1==indices[j+1] else None
    return left,right

def reusable_cls_map(current,prior):
    # Same fine-tuned encoder, source sentence, immediate retained-context neighbors.
    # Never reuse document-level BiLSTM ROLE predictions: their sequence changed.
    if prior is None:return {}
    prev_by_idx={source_idx:j for j,source_idx in enumerate(prior['original_indices'])}
    found={}
    for j,k in enumerate(current['original_indices']):
        old_i=prev_by_idx.get(k)
        if old_i is None:continue
        if current['sentences'][j]!=prior['sentences'][old_i]:continue
        if effective_neighbors(current['original_indices'],j)!=effective_neighbors(prior['original_indices'],old_i):continue
        found[j]=old_i
    return found

@torch.inference_mode()
def encode_case(case):
    sents=case['sentences']
    raw=TOKENIZER(sents,add_special_tokens=False,padding=False,return_attention_mask=False)['input_ids']
    original=case['original_indices']
    reusable=reusable_cls_map(case,PRIOR_120_BY_ID.get(case['id'])) if len(sents)>120 else {}
    emb=np.empty((len(sents),768),dtype=np.float16)
    for j,i_old in reusable.items():
        emb[j]=PRIOR_120_BY_ID[case['id']]['cls'][i_old].astype(np.float16)
    pending=[i for i in range(len(sents)) if i not in reusable]
    for start in range(0,len(pending),ENCODER_BATCH_SIZE):
        ids=pending[start:start+ENCODER_BATCH_SIZE]
        packed=[pack_context_ids(raw[i],
                   raw[i-1] if i and original[i-1]+1==original[i] else [],
                   raw[i+1] if i+1<len(sents) and original[i]+1==original[i+1] else [],
                   max_length=256,target_cap=192) for i in ids]
        batch=COLLATOR(packed)
        batch={k:v.to(DEVICE) for k,v in batch.items()}
        with torch.amp.autocast(device_type='cuda',enabled=(DEVICE.type=='cuda')):
            out=ENCODER.base_model(**batch,return_dict=True).last_hidden_state[:,0,:]
        emb[ids]=out.float().cpu().numpy().astype(np.float16)
    CACHE_STATS['partial_cls_reused_240']+=len(reusable)
    CACHE_STATS['new_cls_encoded']+=len(pending)
    encoded=torch.from_numpy(emb)
    starts=[0]+[i for i in range(1,len(sents)) if original[i]!=original[i-1]+1]
    ends=starts[1:]+[len(sents)]
    role_parts=[]
    for a,b in zip(starts,ends):
        logits=HEAD(encoded[a:b].float().unsqueeze(0).to(DEVICE),
                    torch.tensor([b-a],dtype=torch.long))[0]
        role_parts.append(logits.float().softmax(-1).cpu().numpy().astype(np.float16))
    return encoded.numpy(),np.concatenate(role_parts,axis=0)

def sentence_tfidf_features(sentences):

        vec=TfidfVectorizer(stop_words='english',ngram_range=(1,2),max_features=8000).fit_transform(sentences)

def budgeted_selection(sents,ranking,budget=SUMMARY_WORD_BUDGET,vectors=None):
    selected=[];words=0
    for idx in ranking:
        i=int(idx)
        wc=len(sents[i].split())
        if wc<5 or wc>budget or words+wc>budget:continue
        if vectors is not None and selected:
            if float(cosine_similarity(vectors[i],vectors[selected]).max())>=SUMMARY_REPETITION_THRESHOLD:
                continue
        selected.append(i);words+=wc
        if words>=budget:break
    return sorted(selected)

def lead_indices(sentences,budget=SUMMARY_WORD_BUDGET):
    selected=[];count=0
    for i,s in enumerate(sentences):
        n=len(s.split())
        if n<5:continue
        if count+n>budget:break  # Keep exact Stage 3.4 Lead behavior.
        selected.append(i);count+=n
    return selected

def tfidf_indices(case):
    v,center=case['tfidf_features']
    return budgeted_selection(case['sentences'],np.argsort(-center,kind='stable'),vectors=v)

def create_extract(case,scores,budget=SUMMARY_WORD_BUDGET):
    v,_=case['tfidf_features']
    idx=budgeted_selection(case['sentences'],np.argsort(-scores,kind='stable'),budget,v)
    return '\n'.join(case['sentences'][i] for i in idx),idx

def flatten_train(train_cases,weak_method,with_roles):
    feats=[];labels=[]

        labels.append(case['weak_labels'][weak_method].astype(np.float32))
    return torch.from_numpy(np.concatenate(feats)),torch.from_numpy(np.concatenate(labels))

def full_sha256(path):
    hasher=hashlib.sha256()
    with Path(path).open('rb') as f:
        for chunk in iter(lambda:f.read(4*1024*1024),b''):hasher.update(chunk)
    return hasher.hexdigest()
