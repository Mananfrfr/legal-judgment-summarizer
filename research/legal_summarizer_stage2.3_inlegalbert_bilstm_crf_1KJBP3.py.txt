def make_neighbor_indices(frame):
    """Exactly one previous/next from the SAME doc, ordered by sentence_index."""
    if frame.empty: return [],[]
    keys=frame[['doc_index','sentence_index']].reset_index(drop=True)
    # Preprocessor must already sort sentences within each document.
    assert keys.equals(keys.sort_values(['doc_index','sentence_index']).reset_index(drop=True))
    n=len(frame)
    prev=[i-1 if i>0 and keys.at[i-1,'doc_index']==keys.at[i,'doc_index'] else None for i in range(n)]
    nex=[i+1 if i+1<n and keys.at[i+1,'doc_index']==keys.at[i,'doc_index'] else None for i in range(n)]
    return prev,nex

def pack_context_ids(target_ids, previous_ids, next_ids, max_length=256, target_cap=192):
    """First BERT segment is the target; second consists of balanced neighboring text."""
    assert max_length>=8 and target_cap>=1
    tgt=list(target_ids[:min(target_cap,max_length-3)])
    budget=max_length-len(tgt)-3  # CLS, first SEP, final SEP
    pmark=MARK_PREV if previous_ids else []
    nmark=MARK_NEXT if next_ids else []
    if len(pmark)+len(nmark)>budget:
        raise ValueError('Not enough context-token budget for markers')
    free=budget-len(pmark)-len(nmark)
    prev_cap=min(len(previous_ids), (free+1)//2) if previous_ids else 0
    next_cap=min(len(next_ids), free//2) if next_ids else 0
    # If one neighbor is short/missing, let the other use the unused space.
    leftover=free-prev_cap-next_cap
    prev_cap+=min(len(previous_ids)-prev_cap,leftover)
    leftover=free-prev_cap-next_cap
    next_cap+=min(len(next_ids)-next_cap,leftover)
    prev_part=list(previous_ids[-prev_cap:]) if prev_cap else []  # closest end of previous
    next_part=list(next_ids[:next_cap]) if next_cap else []        # closest start of next
    context=pmark+prev_part+nmark+next_part
    ids=[TOKENIZER.cls_token_id]+tgt+[TOKENIZER.sep_token_id]+context+[TOKENIZER.sep_token_id]
    segment0=len(tgt)+2
    result={'input_ids':ids,'attention_mask':[1]*len(ids),
            'token_type_ids':[0]*segment0+[1]*(len(ids)-segment0)}
    assert len(ids)<=max_length
    assert ids[1:1+len(tgt)]==tgt, 'The target must remain unchanged after truncation'
    return result

class RoleDataset(Dataset):
    def __init__(self, frame, mode, tokenizer=TOKENIZER):
        if mode not in ('target256','context256'): raise ValueError(mode)
        self.mode=mode
        texts=frame.text.astype(str).tolist()
        self.labels=[LABEL2ID[x] for x in frame.label.tolist()]
        if mode=='target256':
            batch=tokenizer(texts,truncation=True,max_length=CFG['max_length'],
                            add_special_tokens=True,padding=False,return_attention_mask=True)
            self.rows=[{key:value[i] for key,value in batch.items()} for i in range(len(texts))]
        else:
            raw_tokens=tokenizer(texts,add_special_tokens=False,padding=False,return_attention_mask=False)['input_ids']
            previous,following=make_neighbor_indices(frame)
            self.rows=[pack_context_ids(raw_tokens[i],
                                        raw_tokens[previous[i]] if previous[i] is not None else [],
                                        raw_tokens[following[i]] if following[i] is not None else [],
                                        max_length=CFG['max_length'],target_cap=CFG['max_target_context'])
                       for i in range(len(texts))]
        assert len(self.rows)==len(self.labels)
    def __len__(self): return len(self.labels)
    def __getitem__(self,i): return {**self.rows[i],'labels':self.labels[i]}

def frame_digest(frame):
    h=hashlib.sha256()
    for doc, start, end, sentence, label in frame[['doc_index','start','end','text','label']].itertuples(index=False,name=None):
        h.update(json.dumps([int(doc),int(start),int(end),str(sentence),str(label)],
                            ensure_ascii=False).encode('utf-8'))
        h.update(b'\n')
    return h.hexdigest()

def checkpoint_signature(path):
    # Quick deterministic fingerprint of the actual finetuned checkpoint, not the base model.
    h=hashlib.sha256()
    for name in ('config.json','tokenizer.json','model.safetensors'):
        p=path/name
        if not p.is_file(): raise FileNotFoundError(p)
        h.update(name.encode()); h.update(str(p.stat().st_size).encode())
        with p.open('rb') as f:
            h.update(f.read(128*1024))
            if p.stat().st_size>128*1024:
                f.seek(-128*1024,2);h.update(f.read(128*1024))
    return h.hexdigest()

@torch.inference_mode()
def encode_split(ds, split):
    print('Encoding',split,'with saved fine-tuned context InLegalBERT checkpoint...')
    loader=DataLoader(ds,batch_size=CFG['encoder_batch_size'],shuffle=False,
                      collate_fn=COLLATOR,num_workers=0)
    vectors=[]
    for batch in tqdm(loader,desc='Frozen InLegalBERT '+split):
        features_in={k:v.to(DEVICE) for k,v in batch.items() if k!='labels'}
        with torch.amp.autocast('cuda'):
            states=ENCODER.base_model(**features_in,return_dict=True).last_hidden_state
        vectors.append(states[:,0,:].float().cpu().to(torch.float16))
    return torch.cat(vectors,dim=0)

def load_or_encode(frame,ds,split):
    path=FEATURE_DIR/f'{split}_context_CLS.pt'
    digest=frame_digest(frame)
    if path.exists() and not FORCE_REEXTRACT_EMBEDDINGS:
        item=torch.load(path,map_location='cpu',weights_only=True)
        if (item.get('data_digest')==digest and item.get('checkpoint_signature')==SIGNATURE
              and item.get('max_length')==CFG['max_length']
              and item.get('max_target_context')==CFG['max_target_context']
              and tuple(item['features'].shape)==(len(frame),768)
              and torch.equal(item['labels'],torch.tensor([LABEL2ID[x] for x in frame.label],dtype=torch.long))):
            print('Reusing verified embedding cache:',path)
            return item['features']
        print('Out-of-date embedding cache; recomputing:',split)
    features=encode_split(ds,split)
    assert features.shape==(len(frame),768) and torch.isfinite(features).all()
    payload={'features':features.contiguous(),
             'labels':torch.tensor([LABEL2ID[x] for x in frame.label],dtype=torch.long),
             'data_digest':digest,'checkpoint_signature':SIGNATURE,
             'max_length':CFG['max_length'], 'max_target_context':CFG['max_target_context']}
    temp=path.with_suffix('.tmp')
    torch.save(payload,temp);temp.replace(path)
    print('Cached',split,features.shape,'at',path)
    return features

class JudgmentSequences(Dataset):
    def __init__(self,frame,embeddings):
        if len(frame)!=len(embeddings): raise ValueError('Feature row mismatch')
        self.frame=frame.reset_index(drop=True)
        self.embeddings=embeddings
        self.parts=[]
        for doc, part in self.frame.groupby('doc_index',sort=True):
            idx=part.index.to_numpy(dtype=np.int64)
            expected=np.arange(len(idx))
            actual=part.sentence_index.to_numpy()
            assert np.array_equal(actual,expected),f'Non-consecutive sentence order in document {doc}'
            self.parts.append(idx)
        assert sum(len(x) for x in self.parts)==len(frame)
    def __len__(self): return len(self.parts)
    def __getitem__(self,i):
        ix=self.parts[i]
        return (self.embeddings[ix].float(),
                torch.tensor([LABEL2ID[x] for x in self.frame.iloc[ix].label],dtype=torch.long),
                torch.tensor(ix,dtype=torch.long))

def collate_judgments(batch):
    b=len(batch);lengths=torch.tensor([len(x[0]) for x in batch],dtype=torch.long)
    maxlen=int(lengths.max());dim=batch[0][0].shape[-1]
    feats=torch.zeros(b,maxlen,dim,dtype=torch.float32)
    labels=torch.zeros(b,maxlen,dtype=torch.long)
    mask=torch.zeros(b,maxlen,dtype=torch.bool)
    orig_index=torch.full((b,maxlen),-1,dtype=torch.long)
    for j,(x,y,indices) in enumerate(batch):
        n=len(x);feats[j,:n]=x;labels[j,:n]=y;mask[j,:n]=True;orig_index[j,:n]=indices
    return dict(features=feats,labels=labels,mask=mask,lengths=lengths,indices=orig_index)

class SentenceBiLSTM(nn.Module):
    def __init__(self,in_dim=768,n_labels=13,hidden=128,dropout=0.30):
        super().__init__()
        self.norm=nn.LayerNorm(in_dim)
        self.drop=nn.Dropout(dropout)
        self.lstm=nn.LSTM(in_dim,hidden,num_layers=1,batch_first=True,bidirectional=True)
        self.classifier=nn.Linear(hidden*2,n_labels)
    def forward(self,x,lengths):
        x=self.drop(self.norm(x))
        packed=nn.utils.rnn.pack_padded_sequence(x,lengths.cpu(),batch_first=True,
                                                  enforce_sorted=False)
        packed_out,_=self.lstm(packed)
        output,_=nn.utils.rnn.pad_packed_sequence(packed_out,batch_first=True,total_length=x.shape[1])
        return self.classifier(self.drop(output))

class LinearChainCRF(nn.Module):
    """From-role to-role transition scores; variable length via boolean mask."""
    def __init__(self,num_tags):
        super().__init__()
        self.num_tags=num_tags
        self.start=nn.Parameter(torch.zeros(num_tags))
        self.end=nn.Parameter(torch.zeros(num_tags))
        self.transitions=nn.Parameter(torch.zeros(num_tags,num_tags))
    def log_partition(self,emissions,mask):
        assert mask[:,0].all()
        scores=self.start+emissions[:,0]
        for t in range(1,emissions.shape[1]):
            step=torch.logsumexp(scores.unsqueeze(2)+self.transitions.unsqueeze(0)
                                  +emissions[:,t].unsqueeze(1),dim=1)
            scores=torch.where(mask[:,t].unsqueeze(1),step,scores)
        return torch.logsumexp(scores+self.end,dim=1)
    def gold_score(self,emissions,tags,mask):
        s=self.start[tags[:,0]]+emissions[:,0].gather(1,tags[:,0:1]).squeeze(1)
        for t in range(1,emissions.shape[1]):
            step=(self.transitions[tags[:,t-1],tags[:,t]]+
                  emissions[:,t].gather(1,tags[:,t:t+1]).squeeze(1))
            s=s+step*mask[:,t]
        last=mask.long().sum(1)-1
        last_tags=tags.gather(1,last.unsqueeze(1)).squeeze(1)
        return s+self.end[last_tags]
    def forward(self,emissions,tags,mask):
        # Mean negative log likelihood per *real sentence*, not per document.
        assert mask.dtype==torch.bool and mask[:,0].all()
        return (self.log_partition(emissions,mask)-self.gold_score(emissions,tags,mask)).sum()/mask.sum()
    @torch.no_grad()
    def decode(self,emissions,mask):
        scores=self.start+emissions[:,0]
        history=[]
        for t in range(1,emissions.shape[1]):
            candidate=scores.unsqueeze(2)+self.transitions.unsqueeze(0)+emissions[:,t].unsqueeze(1)
            best,backptr=candidate.max(dim=1)
            history.append(backptr)
            scores=torch.where(mask[:,t].unsqueeze(1),best,scores)
        best_last=(scores+self.end).argmax(1)
        result=[]
        for b in range(emissions.shape[0]):
            n=int(mask[b].sum().item())
            label=int(best_last[b].item())
            path=[label]
            for t in range(n-2,-1,-1):
                label=int(history[t][b,label].item())
                path.append(label)
            result.append(path[::-1])
        return result

# Tiny brute-force numerical correctness test: two valid time steps, three roles.
import itertools
crf_test=LinearChainCRF(3)
with torch.no_grad():
    crf_test.transitions.copy_(torch.tensor([[.3,.1,-.2],[0.,.2,.5],[-.1,.4,.2]]))
    crf_test.start.copy_(torch.tensor([.1,-.2,.3]));crf_test.end.copy_(torch.tensor([.2,.1,-.1]))
em=torch.tensor([[[.5,.3,-.1],[.1,.8,.0],[90.,-90.,77.]]],requires_grad=True)
mask=torch.tensor([[True,True,False]])
gold=torch.tensor([[0,1,0]])
scores=[]
for first,second in itertools.product(range(3),repeat=2):
    scores.append(crf_test.start[first]+em[0,0,first]+crf_test.transitions[first,second]
                  +em[0,1,second]+crf_test.end[second])
manual_logz=torch.logsumexp(torch.stack(scores),dim=0)
assert torch.allclose(crf_test.log_partition(em,mask)[0],manual_logz,atol=1e-5)
assert torch.allclose(crf_test.gold_score(em,gold,mask)[0],scores[1],atol=1e-5)
assert crf_test.decode(em,mask)[0]==list(np.unravel_index(int(torch.stack(scores).argmax()),(3,3)))
loss=crf_test(em,gold,mask)
loss.backward()
assert torch.isfinite(loss) and torch.isfinite(em.grad).all()
assert em.grad[0,2].abs().sum()==0, 'Masked padding must not influence gradients'
print('CRF brute-force, Viterbi, masking and gradient checks: PASSED.')

def dev_metrics(y_true,y_pred):

                macro_f1=float(f1_score(y_true,y_pred,labels=classes,average='macro',zero_division=0)),

def build_heads(use_crf):
    set_seed(SEED)
    net=SentenceBiLSTM(hidden=CFG['hidden_size'],n_labels=len(LABELS),
                       dropout=CFG['dropout']).to(DEVICE)
    crf=LinearChainCRF(len(LABELS)).to(DEVICE) if use_crf else None
    return net,crf

@torch.no_grad()
def predict_docs(net,crf,docs):
    net.eval()
    if crf is not None:crf.eval()
    loader=DataLoader(docs,batch_size=CFG['doc_batch_size'],shuffle=False,
                      collate_fn=collate_judgments,num_workers=0)
    idxs=[];ys=[];preds=[]
    for batch in loader:
        x=batch['features'].to(DEVICE);lengths=batch['lengths']
        logits=net(x,lengths)
        mask=batch['mask'].to(DEVICE)
        decoded=crf.decode(logits,mask) if crf is not None else logits.argmax(-1).tolist()
        for i,n in enumerate(lengths.tolist()):
            idxs.extend(batch['indices'][i,:n].tolist())
            ys.extend(batch['labels'][i,:n].tolist())
            preds.extend(decoded[i][:n])
    order=np.argsort(np.array(idxs))
    idxs=np.array(idxs)[order]
    ys=np.array(ys)[order];preds=np.array(preds)[order]
    assert np.array_equal(idxs,np.arange(len(docs.frame))),'Prediction order changed'
    assert np.array_equal(ys,np.array([LABEL2ID[x] for x in docs.frame.label]))
    return ys,preds

def train_head(name,use_crf):
    folder=OUTPUT/name
    folder.mkdir(parents=True,exist_ok=True)
    config={'name':name,'use_crf':use_crf,'seed':SEED,'cfg':CFG,
            'feature_checkpoint_signature':SIGNATURE,
            'train_digest':frame_digest(train_df),'dev_digest':frame_digest(dev_df)}
    best_file=folder/'best_head.pt'
    if (not FORCE_RETRAIN and best_file.is_file() and (folder/'dev_metrics.json').is_file()
        and (folder/'head_config.json').is_file() and (folder/'training_complete.json').is_file()
        and json.loads((folder/'head_config.json').read_text())==config):
        print('Using completed, verified previous run for',name)
        return json.loads((folder/'dev_metrics.json').read_text())
    print('\n'+'='*55+'\nTRAIN',name,'\n'+'='*55)
    net,crf=build_heads(use_crf)
    optimizer=torch.optim.AdamW(list(net.parameters())+(list(crf.parameters()) if crf is not None else []),
                                 lr=CFG['lr'],weight_decay=CFG['weight_decay'])
    best=-1.0;best_epoch=None;stale=0;history=[]
    for epoch in range(1,CFG['epochs']+1):
        net.train()
        if crf is not None:crf.train()
        loader=DataLoader(TRAIN_DOCS,batch_size=CFG['doc_batch_size'],shuffle=True,
                          collate_fn=collate_judgments,num_workers=0,
                          generator=torch.Generator().manual_seed(SEED+epoch))
        cum_loss=0.0;n_total=0
        progress=tqdm(loader,desc=f'{name} epoch {epoch}/{CFG["epochs"]}',leave=False)
        for batch in progress:
            x=batch['features'].to(DEVICE)
            gold=batch['labels'].to(DEVICE)
            mask=batch['mask'].to(DEVICE)
            emissions=net(x,batch['lengths'])
            if crf is not None:
                loss=crf(emissions,gold,mask)
            else:
                loss=nn.functional.cross_entropy(emissions[mask],gold[mask])
            if not torch.isfinite(loss): raise RuntimeError(f'Nonfinite loss in {name}')
            optimizer.zero_grad(set_to_none=True)
            loss.backward()
            nn.utils.clip_grad_norm_(list(net.parameters())+
                (list(crf.parameters()) if crf is not None else []),CFG['max_grad_norm'])
            optimizer.step()
            count=int(mask.sum())
            cum_loss+=float(loss.item())*count;n_total+=count
        y,p=predict_docs(net,crf,DEV_DOCS)
        metrics=dev_metrics(y,p)
        record=dict(epoch=epoch,train_token_loss=cum_loss/n_total,**metrics)
        history.append(record)
        pd.DataFrame(history).to_csv(folder/'training_history.csv',index=False)
        print(name, 'epoch',epoch,'loss',round(record['train_token_loss'],4),
              'Macro F1',round(metrics['macro_f1'],4),
              'Weighted F1',round(metrics['weighted_f1'],4),
              'accuracy',round(metrics['accuracy'],4))
        if metrics['macro_f1']>best+1e-7:
            best=metrics['macro_f1'];best_epoch=epoch;stale=0
            torch.save({'bilstm':net.state_dict(),
                        'crf':crf.state_dict() if crf is not None else None,
                        'epoch':epoch,'feature_signature':SIGNATURE,'config':config},best_file)
            (folder/'head_config.json').write_text(json.dumps(config,indent=2))
            save_dev_files(folder,name,y,p,metrics)
            (folder/'best_epoch.json').write_text(json.dumps({'epoch':epoch,**metrics},indent=2))
            print('   Saved improved DEV Macro F1 checkpoint.')
        else:
            stale+=1
            if stale>=CFG['patience']:
                print('   Early stopping; best epoch:',best_epoch)
                break
    (folder/'training_complete.json').write_text(json.dumps(
        {'finished':True,'epochs_run':len(history),'best_epoch':best_epoch,'best_dev_macro_f1':best},indent=2))
    return json.loads((folder/'dev_metrics.json').read_text())
